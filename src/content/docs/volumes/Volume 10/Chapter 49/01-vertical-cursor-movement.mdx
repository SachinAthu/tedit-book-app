## Implementation Goal

This first implementation milestone adds Up and Down Cursor movement with preferred-column preservation, following ADR-014.

For this milestone, the Cursor must:

- Move between logical lines using Up and Down operations.
- Stop at the first and last lines without changing position.
- Clamp its actual column when moving to a shorter line.
- Restore its preferred column when moving back to a longer line.
- Preserve the existing Left and Right movement behavior.

```c
/* src/cursor/cursor.c */

#include "cursor.h"

void cursor_init(Cursor *cursor)
{
    if (cursor == NULL)
    {
        return;
    }

    cursor->row = 0;
    cursor->column = 0;
    cursor->preferred_column = 0;
}

bool cursor_set_position(
    Cursor *cursor,
    const Document *document,
    size_t row,
    size_t column
)
{
    if (cursor == NULL || document == NULL ||
        row >= document->line_count)
    {
        return false;
    }

    const DocumentLine *line = &document->lines[row];

    if (column > line->length)
    {
        return false;
    }

    cursor->row = row;
    cursor->column = column;
    cursor->preferred_column = column;

    return true;
}

bool cursor_advance(Cursor *cursor, const Document *document)
{
    if (cursor == NULL || document == NULL ||
        cursor->row >= document->line_count)
    {
        return false;
    }

    const DocumentLine *line = &document->lines[cursor->row];

    if (cursor->column >= line->length)
    {
        return false;
    }

    cursor->column += 1;
    cursor->preferred_column = cursor->column;

    return true;
}

bool cursor_move_left(Cursor *cursor)
{
    if (cursor == NULL || cursor->column == 0)
    {
        return false;
    }

    cursor->column -= 1;
    cursor->preferred_column = cursor->column;

    return true;
}

bool cursor_move_right(Cursor *cursor, const Document *document)
{
    if (cursor == NULL || document == NULL ||
        cursor->row >= document->line_count)
    {
        return false;
    }

    const DocumentLine *line = &document->lines[cursor->row];

    if (cursor->column >= line->length)
    {
        return false;
    }

    cursor->column += 1;
    cursor->preferred_column = cursor->column;

    return true;
}

bool cursor_move_up(Cursor *cursor, const Document *document)
{
    if (cursor == NULL || document == NULL ||
        cursor->row >= document->line_count ||
        cursor->row == 0)
    {
        return false;
    }

    const size_t target_row = cursor->row - 1;
    const DocumentLine *target_line = &document->lines[target_row];

    cursor->row = target_row;
    cursor->column = cursor->preferred_column;

    if (cursor->column > target_line->length)
    {
        cursor->column = target_line->length;
    }

    return true;
}

bool cursor_move_down(Cursor *cursor, const Document *document)
{
    if (cursor == NULL || document == NULL ||
        cursor->row >= document->line_count ||
        cursor->row + 1 >= document->line_count)
    {
        return false;
    }

    const size_t target_row = cursor->row + 1;
    const DocumentLine *target_line = &document->lines[target_row];

    cursor->row = target_row;
    cursor->column = cursor->preferred_column;

    if (cursor->column > target_line->length)
    {
        cursor->column = target_line->length;
    }

    return true;
}
```

static void test_cursor_vertical_movement_preserves_preferred_column(void)
{
    Document document;
    Cursor cursor;

    assert(document_init(&document));

    /* Build lines with lengths 6, 2, 0, and 6. */
    for (size_t column = 0; column < 6; ++column)
    {
        assert(document_insert_character(
            &document, 0, column, 'a'));
    }

    assert(document_split_line(&document, 0, 6));
    assert(document_insert_character(&document, 1, 0, 'b'));
    assert(document_insert_character(&document, 1, 1, 'b'));
    assert(document_split_line(&document, 1, 2));
    assert(document_split_line(&document, 2, 0));

    for (size_t column = 0; column < 6; ++column)
    {
        assert(document_insert_character(
            &document, 3, column, 'd'));
    }

    cursor_init(&cursor);
    assert(cursor_set_position(&cursor, &document, 0, 5));
    assert(cursor.preferred_column == 5);

    assert(cursor_move_down(&cursor, &document));
    assert(cursor.row == 1);
    assert(cursor.column == 2);
    assert(cursor.preferred_column == 5);

    assert(cursor_move_down(&cursor, &document));
    assert(cursor.row == 2);
    assert(cursor.column == 0);
    assert(cursor.preferred_column == 5);

    assert(cursor_move_down(&cursor, &document));
    assert(cursor.row == 3);
    assert(cursor.column == 5);
    assert(cursor.preferred_column == 5);

    assert(!cursor_move_down(&cursor, &document));
    assert(cursor.row == 3 && cursor.column == 5);

    assert(cursor_move_up(&cursor, &document));
    assert(cursor.row == 2 && cursor.column == 0);
    assert(cursor.preferred_column == 5);

    assert(cursor_move_up(&cursor, &document));
    assert(cursor.row == 1 && cursor.column == 2);
    assert(cursor.preferred_column == 5);

    assert(cursor_move_up(&cursor, &document));
    assert(cursor.row == 0 && cursor.column == 5);
    assert(cursor.preferred_column == 5);

    assert(!cursor_move_up(&cursor, &document));
    assert(cursor.row == 0 && cursor.column == 5);

    document_destroy(&document);
}

## How the implementation works

The `Cursor` structure now contains three fields:

| Field              | Purpose                                |
| ------------------ | -------------------------------------- |
| `row`              | Current logical line                   |
| `column`           | Current position within that line      |
| `preferred_column` | Desired column for vertical navigation |

The two new functions, `cursor_move_up()` and `cursor_move_down()`, use the Document to validate the current row and locate the destination line. They preserve `preferred_column` while limiting `column` to the destination line's length.

For example, moving down from column 5 through lines of lengths 2, 0, and 6 produces:

| Destination line | Actual column | Preferred column |
| ---------------- | ------------- | ---------------- |
| Length 2         | 2             | 5                |
| Empty line       | 0             | 5                |
| Length 6         | 5             | 5                |

Horizontal movement resets the preferred column to the new actual column. This prevents a previous vertical-navigation target from unexpectedly affecting subsequent movement after the user moves left or right.

## Implementation invariants

The new Cursor operations maintain these rules:

- A successful vertical movement changes the row by exactly one.
- Vertical movement does not change the preferred column.
- The actual column never exceeds the destination line's length.
- Movement beyond the first or last line returns `false` without changing the Cursor.
- Invalid arguments are rejected without modifying the Cursor.
- Horizontal movement updates the preferred column when movement succeeds.

One integration detail remains: `main.c` still handles keyboard input directly, and some editing operations assign `cursor.column` directly. Before enabling vertical navigation interactively, those paths must keep `preferred_column` consistent with insertion, deletion, splitting, and merging.

static void test_document_insert_and_delete(void)
{
    Document document;
    DocumentLine *line;

    assert(document_init(&document));
    assert(document_insert_character(&document, 0, 0, 'a'));
    assert(document_insert_character(&document, 0, 1, 'c'));
    assert(document_insert_character(&document, 0, 1, 'b'));

    line = document_get_line(&document, 0);
    assert(line != NULL);
    assert(line->length == 3);
    assert(memcmp(line->data, "abc", 3) == 0);

    assert(document_delete_character(&document, 0, 1));
    assert(line->length == 2);
    assert(memcmp(line->data, "ac", 2) == 0);

    assert(!document_delete_character(&document, 0, 2));
    assert(!document_delete_character(&document, 1, 0));
    assert(line->length == 2);

static void test_document_merge_lines(void)
{

    assert(document_insert_character(&document, 0, 1, 'b'));
    assert(document_split_line(&document, 0, 2));

static void test_line_editing_cursor_transitions(void)
{
    EditorState state;
    const DocumentLine *previous_line;

    assert(editor_state_init(&state));
    assert(document_insert_character(&state.document, 0, 0, 'a'));
    assert(document_insert_character(&state.document, 0, 1, 'b'));
    assert(cursor_set_position(&state.cursor, &state.document, 0, 1));

    assert(document_split_line(
        &state.document, state.cursor.row, state.cursor.column));
    state.cursor.row += 1;
    state.cursor.column = 0;
    state.is_dirty = true;
    assert(state.cursor.row == 1 && state.cursor.column == 0);
    assert(state.document.line_count == 2);

    assert(document_insert_character(&state.document, 1, 0, 'c'));
    assert(cursor_set_position(&state.cursor, &state.document, 1, 0));
    previous_line = document_get_line(&state.document, 0);
    assert(previous_line != NULL);
    state.cursor.row -= 1;
    state.cursor.column = previous_line->length;
    assert(document_merge_lines(&state.document, state.cursor.row));
    assert(state.cursor.row == 0 && state.cursor.column == 1);
    assert(state.document.line_count == 1);
    assert(state.is_dirty);

    editor_state_destroy(&state);
}
