## Dependency correction implemented

The refactor is complete in the working repository. The previous dependency was:

```
Editor State ──► File Manager
```

The corrected dependency is:

```
Application Layer
  Command Engine
      │
      ├──────────► File Manager
      │
      ▼
  Editor State
```

The Command Engine now coordinates file operations and the resulting state transitions. The Core Layer no longer calls file-system operations directly.

## Source changes

| File                           | Change                                                                                                                                                             |
| ------------------------------ | ------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| `src/state/editor_state.h`     | Removed file-operation APIs and introduced state-transition APIs.                                                                                                  |
| `src/state/editor_state.c`     | Removed the File Manager and logger dependencies; added operations for adopting a loaded document, transferring file-path ownership, and marking a document clean. |
| `src/command/command_engine.h` | Added application-level load, save, and Save As APIs.                                                                                                              |
| `src/command/command_engine.c` | Added transactional file-operation coordination and retained the existing save-failure guarantees.                                                                 |
| `src/main.c`                   | Routed startup-file loading through the Command Engine.                                                                                                            |
| `tests/test_file.c`            | Updated state-level file-loading tests to use the application coordinator.                                                                                         |
| `tests/test_file_save.c`       | Updated state-level save tests to use the application coordinator.                                                                                                 |
| `tests/test_command_engine.c`  | Updated the Save As test to use the application coordinator.                                                                                                       |

The File Manager still owns UTF-8 decoding, serialization, temporary-file creation, and atomic replacement. The Editor State still owns the document, file association, cursor, viewport, and Dirty Flag.

The refactor preserves the important transaction guarantees: failed loads leave the existing editing state unchanged, failed saves preserve the Dirty Flag, and failed Save As operations preserve the previous file association.

```c
/* src/state/editor_state.c */

#include "editor_state.h"
#include <string.h>
#include <stdlib.h>

bool editor_state_init(EditorState *state)
{
    if (state == NULL)
    {
        return false;
    }

    /* Zero every field first so partial failures leave a clean state. */
    memset(state, 0, sizeof(*state));

    if (!document_init(&state->document))
    {
        return false;
    }

    cursor_init(&state->cursor);
    viewport_init(&state->viewport);
    state->configuration.tab_width = DEFAULT_TAB_WIDTH;
    state->is_dirty = false;
    state->file_path = NULL;
    state->message[0] = '\0';
    state->filename_entry_active = false;
    state->filename_entry[0] = '\0';
    state->filename_entry_length = 0;
    state->filename_entry_cursor = 0;
    state->quit_confirmation_active = false;
    state->quit_after_save = false;

    return true;
}

bool editor_state_adopt_loaded_document(EditorState *state,
                                        Document *replacement,
                                        char *owned_file_path)
{
    if (state == NULL || replacement == NULL || owned_file_path == NULL)
    {
        return false;
    }

    document_destroy(&state->document);
    state->document = *replacement;
    memset(replacement, 0, sizeof(*replacement));

    free(state->file_path);
    state->file_path = owned_file_path;
    cursor_init(&state->cursor);
    viewport_init(&state->viewport);
    state->is_dirty = false;
    state->filename_entry_active = false;
    state->filename_entry[0] = '\0';
    state->filename_entry_length = 0;
    state->filename_entry_cursor = 0;
    state->quit_confirmation_active = false;
    state->quit_after_save = false;
    return true;
}

void editor_state_take_file_path(EditorState *state, char *owned_file_path)
{
    if (state == NULL)
    {
        free(owned_file_path);
        return;
    }

    if (state->file_path != owned_file_path)
    {
        free(state->file_path);
        state->file_path = owned_file_path;
    }
}

void editor_state_mark_clean(EditorState *state)
{
    if (state != NULL)
    {
        state->is_dirty = false;
    }
}

bool editor_state_set_message(EditorState *state, const char *message)
{
    size_t length;

    if (state == NULL || message == NULL)
    {
        return false;
    }

    /* Copy at most a full buffer's worth, leaving room for the terminator. */
    length = strlen(message);
    if (length >= sizeof(state->message))
    {
        length = sizeof(state->message) - 1;
    }

    memmove(state->message, message, length);
    state->message[length] = '\0';
    return true;
}

void editor_state_destroy(EditorState *state)
{
    if (state == NULL)
    {
        return;
    }

    document_destroy(&state->document);
    free(state->file_path);
    state->file_path = NULL;
    cursor_init(&state->cursor);
    viewport_init(&state->viewport);
    state->is_dirty = false;
    state->message[0] = '\0';
    state->filename_entry_active = false;
    state->filename_entry[0] = '\0';
    state->filename_entry_length = 0;
    state->filename_entry_cursor = 0;
    state->quit_confirmation_active = false;
    state->quit_after_save = false;
}
```

### `src/command/command_engine.c`

```c
/* src/command/command_engine.c */

#include "command/command_engine.h"
#include "common/logger.h"

#include "cursor/cursor.h"
#include "document/document.h"
#include "file/file.h"
#include "state/editor_state.h"

#include <stdlib.h>
#include <string.h>

bool command_engine_load_file(EditorState *state, const char *path)
{
    Document replacement;
    char *owned_path;
    size_t path_length;
    FileLoadResult result;

    if (state == NULL || path == NULL || path[0] == '\0')
    {
        return false;
    }

    path_length = strlen(path);
    if (path_length == SIZE_MAX)
    {
        return false;
    }
    owned_path = malloc(path_length + 1);
    if (owned_path == NULL)
    {
        return false;
    }
    memcpy(owned_path, path, path_length + 1);

    if (!document_init(&replacement))
    {
        free(owned_path);
        return false;
    }

    result = file_load_document(path, &replacement);
    if (result != FILE_LOAD_OK)
    {
        TEDIT_LOG_WARN("file", "load failed: path=\"%s\" result=%d", path, (int)result);
        document_destroy(&replacement);
        free(owned_path);
        return false;
    }

    if (!editor_state_adopt_loaded_document(state, &replacement, owned_path))
    {
        document_destroy(&replacement);
        free(owned_path);
        return false;
    }
    TEDIT_LOG_INFO("file", "loaded file: path=\"%s\"", path);
    return true;
}

bool command_engine_save_file(EditorState *state, const char *path)
{
    FileSaveResult result;

    if (state == NULL || path == NULL || path[0] == '\0')
    {
        return false;
    }

    result = file_save_document(path, &state->document);
    if (result != FILE_SAVE_OK)
    {
        TEDIT_LOG_WARN("file", "save failed: path=\"%s\" result=%d dirty=%s",
                       path, (int)result, state->is_dirty ? "true" : "false");
        return false;
    }

    TEDIT_LOG_INFO("file", "saved file: path=\"%s\"", path);
    if (state->is_dirty)
    {
        TEDIT_LOG_INFO("state", "Dirty Flag changed: dirty -> clean after save");
    }
    editor_state_mark_clean(state);
    return true;
}

bool command_engine_save_as(EditorState *state, const char *path)
{
    char *owned_path;
    size_t path_length;
    FileSaveResult result;

    if (state == NULL || path == NULL || path[0] == '\0')
    {
        return false;
    }

    path_length = strlen(path);
    if (path_length == SIZE_MAX)
    {
        return false;
    }
    owned_path = malloc(path_length + 1);
    if (owned_path == NULL)
    {
        return false;
    }
    memcpy(owned_path, path, path_length + 1);

    result = file_save_document(path, &state->document);
    if (result != FILE_SAVE_OK)
    {
        TEDIT_LOG_WARN("file", "save failed: path=\"%s\" result=%d dirty=%s",
                       path, (int)result, state->is_dirty ? "true" : "false");
        free(owned_path);
        return false;
    }

    TEDIT_LOG_INFO("file", "saved file: path=\"%s\"", path);
    editor_state_take_file_path(state, owned_path);
    if (state->is_dirty)
    {
        TEDIT_LOG_INFO("state", "Dirty Flag changed: dirty -> clean after save");
    }
    editor_state_mark_clean(state);
    return true;
}

static void command_mark_document_dirty(EditorState *state)
{
    if (state != NULL && !state->is_dirty)
    {
        TEDIT_LOG_INFO("state", "Dirty Flag changed: clean -> dirty");
    }
    if (state != NULL)
    {
        state->is_dirty = true;
    }
}

static bool command_backspace(EditorState *state, bool *changed)
{
    if (state->cursor.column > 0)
    {
        const size_t delete_column = state->cursor.column - 1;
        if (!document_delete_character(&state->document, state->cursor.row, delete_column))
        {
            return true;
        }
        if (!cursor_set_position(&state->cursor, &state->document,
                                 state->cursor.row, delete_column))
        {
            return false;
        }
    }
    else if (state->cursor.row > 0)
    {
        const size_t previous_row = state->cursor.row - 1;
        const DocumentLine *previous_line = document_get_line(&state->document, previous_row);
        if (previous_line == NULL)
        {
            return true;
        }

        const size_t previous_length = previous_line->length;
        if (!document_merge_lines(&state->document, previous_row))
        {
            return true;
        }
        if (!cursor_set_position(&state->cursor, &state->document, previous_row, previous_length))
        {
            return false;
        }
    }
    else
    {
        return true;
    }

    command_mark_document_dirty(state);
    *changed = true;
    return true;
}

static bool command_delete_character(EditorState *state, bool *changed)
{
    const DocumentLine *line = document_get_line(&state->document, state->cursor.row);

    if (line == NULL)
    {
        return true;
    }

    if (state->cursor.column < line->length)
    {
        if (!document_delete_character(&state->document, state->cursor.row,
                                       state->cursor.column))
        {
            return true;
        }
    }
    else if (state->cursor.row + 1 < state->document.line_count)
    {
        if (!document_merge_lines(&state->document, state->cursor.row))
        {
            return true;
        }
    }
    else
    {
        return true;
    }

    if (!cursor_set_position(&state->cursor, &state->document,
                             state->cursor.row, state->cursor.column))
    {
        return false;
    }

    command_mark_document_dirty(state);
    *changed = true;
    return true;
}

static size_t encode_utf8(uint32_t character, unsigned char bytes[4])
{
    if (character <= 0x7F)
    {
        bytes[0] = (unsigned char)character;
        return 1;
    }
    if (character <= 0x7FF)
    {
        bytes[0] = (unsigned char)(0xC0 | (character >> 6));
        bytes[1] = (unsigned char)(0x80 | (character & 0x3F));
        return 2;
    }
    if (character <= 0xFFFF && !(character >= 0xD800 && character <= 0xDFFF))
    {
        bytes[0] = (unsigned char)(0xE0 | (character >> 12));
        bytes[1] = (unsigned char)(0x80 | ((character >> 6) & 0x3F));
        bytes[2] = (unsigned char)(0x80 | (character & 0x3F));
        return 3;
    }
    if (character <= 0x10FFFF)
    {
        bytes[0] = (unsigned char)(0xF0 | (character >> 18));
        bytes[1] = (unsigned char)(0x80 | ((character >> 12) & 0x3F));
        bytes[2] = (unsigned char)(0x80 | ((character >> 6) & 0x3F));
        bytes[3] = (unsigned char)(0x80 | (character & 0x3F));
        return 4;
    }
    return 0;
}

static size_t filename_previous_boundary(const char *text, size_t position)
{
    if (position == 0)
    {
        return 0;
    }

    --position;
    while (position > 0 && (((unsigned char)text[position] & 0xC0) == 0x80))
    {
        --position;
    }
    return position;
}

static size_t filename_next_boundary(const char *text, size_t length, size_t position)
{
    if (position >= length)
    {
        return length;
    }

    ++position;
    while (position < length && (((unsigned char)text[position] & 0xC0) == 0x80))
    {
        ++position;
    }
    return position;
}

static void filename_entry_clear(EditorState *state)
{
    state->filename_entry_active = false;
    state->filename_entry_length = 0;
    state->filename_entry_cursor = 0;
    state->filename_entry[0] = '\0';
}

static void filename_entry_start(EditorState *state, bool quit_after_save)
{
    state->filename_entry_active = true;
    state->filename_entry_length = 0;
    state->filename_entry_cursor = 0;
    state->filename_entry[0] = '\0';
    state->quit_after_save = quit_after_save;
    (void)editor_state_set_message(state, "Enter: save  Esc: cancel");
}

static bool command_save(EditorState *state, const char *save_path, bool quitting, bool *redraw, bool *should_quit)
{
    const char *current_path = state->file_path != NULL ? state->file_path : save_path;

    if (current_path == NULL)
    {
        filename_entry_start(state, quitting);
        *redraw = true;
        return true;
    }

    if (command_engine_save_file(state, current_path))
    {
        (void)editor_state_set_message(state, "Saved successfully.");
        state->quit_after_save = false;
        if (quitting)
        {
            *should_quit = true;
        }
    }
    else
    {
        (void)editor_state_set_message(state, "Save failed; document remains modified.");
        state->quit_after_save = false;
    }

    *redraw = true;
    return true;
}

static bool filename_entry_execute(EditorState *state, const EditorCommand *command, bool *redraw, bool *should_quit)
{
    if (command->type == EDITOR_COMMAND_CANCEL)
    {
        const bool was_quitting = state->quit_after_save;
        filename_entry_clear(state);
        state->quit_after_save = false;
        (void)editor_state_set_message(state, was_quitting ? "Quit cancelled." : "Save As cancelled.");
        *redraw = true;
        return true;
    }

    if (command->type == EDITOR_COMMAND_MOVE_LEFT)
    {
        const size_t previous = filename_previous_boundary(state->filename_entry,
                                                           state->filename_entry_cursor);
        if (previous != state->filename_entry_cursor)
        {
            state->filename_entry_cursor = previous;
            *redraw = true;
        }
        return true;
    }

    if (command->type == EDITOR_COMMAND_MOVE_RIGHT)
    {
        const size_t next = filename_next_boundary(state->filename_entry,
                                                   state->filename_entry_length,
                                                   state->filename_entry_cursor);
        if (next != state->filename_entry_cursor)
        {
            state->filename_entry_cursor = next;
            *redraw = true;
        }
        return true;
    }

    if (command->type == EDITOR_COMMAND_BACKSPACE)
    {
        if (state->filename_entry_cursor > 0)
        {
            const size_t previous = filename_previous_boundary(state->filename_entry, state->filename_entry_cursor);
            const size_t removed = state->filename_entry_cursor - previous;
            memmove(state->filename_entry + previous,
                    state->filename_entry + state->filename_entry_cursor,
                    state->filename_entry_length - state->filename_entry_cursor + 1);
            state->filename_entry_length -= removed;
            state->filename_entry_cursor = previous;
            *redraw = true;
        }
        return true;
    }

    if (command->type == EDITOR_COMMAND_DELETE_CHARACTER)
    {
        if (state->filename_entry_cursor < state->filename_entry_length)
        {
            const size_t next = filename_next_boundary(state->filename_entry,
                                                       state->filename_entry_length,
                                                       state->filename_entry_cursor);
            const size_t removed = next - state->filename_entry_cursor;
            memmove(state->filename_entry + state->filename_entry_cursor,
                    state->filename_entry + next,
                    state->filename_entry_length - next + 1);
            state->filename_entry_length -= removed;
            *redraw = true;
        }
        return true;
    }

    if (command->type == EDITOR_COMMAND_INSERT_CHARACTER)
    {
        unsigned char encoded[4];
        const size_t encoded_length = encode_utf8(command->character, encoded);

        if (encoded_length == 0)
        {
            return true;
        }
        if (encoded_length > EDITOR_FILENAME_CAPACITY - 1 - state->filename_entry_length)
        {
            (void)editor_state_set_message(state, "Filename is too long.");
            *redraw = true;
            return true;
        }

        memmove(state->filename_entry + state->filename_entry_cursor + encoded_length,
                state->filename_entry + state->filename_entry_cursor,
                state->filename_entry_length - state->filename_entry_cursor + 1);
        memcpy(state->filename_entry + state->filename_entry_cursor, encoded, encoded_length);
        state->filename_entry_length += encoded_length;
        state->filename_entry_cursor += encoded_length;
        (void)editor_state_set_message(state, "Enter: save  Esc: cancel");
        *redraw = true;
        return true;
    }

    if (command->type == EDITOR_COMMAND_INSERT_NEWLINE)
    {
        if (state->filename_entry_length == 0)
        {
            (void)editor_state_set_message(state, "Enter a file name.");
            *redraw = true;
            return true;
        }

        if (command_engine_save_as(state, state->filename_entry))
        {
            const bool should_quit_after_save = state->quit_after_save;
            filename_entry_clear(state);
            state->quit_after_save = false;
            (void)editor_state_set_message(state, "Saved successfully.");
            if (should_quit_after_save)
            {
                *should_quit = true;
            }
        }
        else
        {
            (void)editor_state_set_message(
                state, "Save failed; check path and try again, or Esc to cancel.");
        }
        *redraw = true;
        return true;
    }

    /* Ignore commands not meaningful while a filename is being entered. */
    return true;
}

static bool quit_confirmation_execute(EditorState *state, const EditorCommand *command, const char *save_path, bool *redraw, bool *should_quit)
{
    if (command->type == EDITOR_COMMAND_CANCEL)
    {
        state->quit_confirmation_active = false;
        (void)editor_state_set_message(state, "Quit cancelled.");
        *redraw = true;
        return true;
    }

    if (command->type != EDITOR_COMMAND_INSERT_CHARACTER)
    {
        return true;
    }

    if (command->character == 'n' || command->character == 'N')
    {
        state->quit_confirmation_active = false;
        state->quit_after_save = false;
        *should_quit = true;
        return true;
    }

    if (command->character == 'y' || command->character == 'Y')
    {
        state->quit_confirmation_active = false;
        return command_save(state, save_path, true, redraw, should_quit);
    }

    return true;
}

bool command_engine_execute(EditorState *state, const EditorCommand *command, const char *save_path, bool *redraw, bool *should_quit)
{
    bool changed = false;

    if (state == NULL || command == NULL || redraw == NULL || should_quit == NULL)
    {
        return false;
    }

    *redraw = false;
    *should_quit = false;

    if (state->filename_entry_active)
    {
        return filename_entry_execute(state, command, redraw, should_quit);
    }

    if (state->quit_confirmation_active)
    {
        return quit_confirmation_execute(state, command, save_path, redraw, should_quit);
    }

    switch (command->type)
    {
    case EDITOR_COMMAND_NONE:
        break;
    case EDITOR_COMMAND_QUIT:
        if (state->is_dirty)
        {
            state->quit_confirmation_active = true;
            state->quit_after_save = false;
            (void)editor_state_set_message(state, "Save changes before quitting?");
            *redraw = true;
        }
        else
        {
            *should_quit = true;
        }
        break;
    case EDITOR_COMMAND_SAVE:
        return command_save(state, save_path, false, redraw, should_quit);
    case EDITOR_COMMAND_CANCEL:
        break;
    case EDITOR_COMMAND_INSERT_CHARACTER:
    case EDITOR_COMMAND_INSERT_TAB:
    {
        const uint32_t character = command->type == EDITOR_COMMAND_INSERT_TAB
                                       ? (uint32_t)'\t'
                                       : command->character;
        if (!document_insert_character(&state->document, state->cursor.row, state->cursor.column, character))
        {
            return true;
        }
        if (!cursor_advance(&state->cursor, &state->document))
        {
            return false;
        }
        command_mark_document_dirty(state);
        changed = true;
        break;
    }
    case EDITOR_COMMAND_BACKSPACE:
        if (!command_backspace(state, &changed))
        {
            return false;
        }
        break;
    case EDITOR_COMMAND_INSERT_NEWLINE:
        if (!document_split_line(&state->document, state->cursor.row, state->cursor.column))
        {
            return true;
        }
        if (!cursor_set_position(&state->cursor, &state->document, state->cursor.row + 1, 0))
        {
            return false;
        }
        command_mark_document_dirty(state);
        changed = true;
        break;
    case EDITOR_COMMAND_MOVE_LEFT:
        changed = cursor_move_left(&state->cursor);
        break;
    case EDITOR_COMMAND_MOVE_RIGHT:
        changed = cursor_move_right(&state->cursor, &state->document);
        break;
    case EDITOR_COMMAND_MOVE_UP:
        changed = cursor_move_up(&state->cursor, &state->document, state->configuration.tab_width);
        break;
    case EDITOR_COMMAND_MOVE_DOWN:
        changed = cursor_move_down(&state->cursor, &state->document, state->configuration.tab_width);
        break;
    case EDITOR_COMMAND_PAGE_UP:
        changed = cursor_move_page_up(&state->cursor, &state->document, command->page_rows, state->configuration.tab_width);
        break;
    case EDITOR_COMMAND_PAGE_DOWN:
        changed = cursor_move_page_down(&state->cursor, &state->document, command->page_rows, state->configuration.tab_width);
        break;
    case EDITOR_COMMAND_MOVE_HOME:
        changed = cursor_move_home(&state->cursor);
        break;
    case EDITOR_COMMAND_MOVE_END:
        changed = cursor_move_end(&state->cursor, &state->document);
        break;
    case EDITOR_COMMAND_DELETE_CHARACTER:
        if (!command_delete_character(state, &changed))
        {
            return false;
        }
        break;
    default:
        return false;
    }

    switch (command->type)
    {
    case EDITOR_COMMAND_INSERT_CHARACTER:
    case EDITOR_COMMAND_INSERT_TAB:
    case EDITOR_COMMAND_BACKSPACE:
    case EDITOR_COMMAND_INSERT_NEWLINE:
    case EDITOR_COMMAND_MOVE_LEFT:
    case EDITOR_COMMAND_MOVE_RIGHT:
    case EDITOR_COMMAND_MOVE_HOME:
    case EDITOR_COMMAND_MOVE_END:
    case EDITOR_COMMAND_DELETE_CHARACTER:
        state->cursor.preferred_visual_column_valid = false;
        break;
    default:
        break;
    }

    *redraw = *redraw || changed;
    return true;
}
```

```c
/*
 * How long to wait for the bytes that may follow an ESC key press before
 * deciding the user pressed Escape alone. Long enough for terminals to
 * deliver the rest of an arrow-key sequence, short enough that a lone ESC
 * press still feels instant.

/* Signal handlers only set flags; all terminal and allocation work remains in

static volatile sig_atomic_t termination_requested = 0;

/*
 * Shared handler for the signals we install. A signal can arrive between any
 * two machine instructions, so it must do almost nothing: it only sets a
 * `volatile sig_atomic_t` flag, which the C standard guarantees is safe to
 * modify from an asynchronous handler. The main loop checks these flags and
 * performs the actual work (resize, cleanup, exit) at a safe point.
 */
static void application_signal_handler(int signal_number)
{
    if (signal_number == SIGWINCH)
    {
        resize_requested = 1;
    }
    else
    {
        termination_requested = 1;
    }
}

/*
 * Install handlers for the signals the editor must react to: window resize
 * (SIGWINCH) and termination (SIGINT, SIGTERM, SIGHUP). The previous handler
 * for each signal is saved so it can be restored on exit. If installing one
 * handler fails, any already-installed handlers are rolled back before
 * returning false.
 */
static bool application_install_signal_handlers(void)
{
    struct sigaction action;

    memset(&action, 0, sizeof(action));
    action.sa_handler = application_signal_handler;
    sigemptyset(&action.sa_mask);

    if (sigaction(SIGWINCH, &action, &previous_winch_action) == -1)
    {
        TEDIT_LOG_ERROR("lifecycle", "failed to install SIGWINCH handler");
        return false;
    }
    if (sigaction(SIGINT, &action, &previous_int_action) == -1)
    {
        (void)sigaction(SIGWINCH, &previous_winch_action, NULL);
        return false;
    }
    if (sigaction(SIGTERM, &action, &previous_term_action) == -1)
    {
        (void)sigaction(SIGINT, &previous_int_action, NULL);
        (void)sigaction(SIGWINCH, &previous_winch_action, NULL);
        return false;
    }
    if (sigaction(SIGHUP, &action, &previous_hup_action) == -1)
    {
        (void)sigaction(SIGTERM, &previous_term_action, NULL);
        (void)sigaction(SIGINT, &previous_int_action, NULL);
        (void)sigaction(SIGWINCH, &previous_winch_action, NULL);
        return false;
    }

    signal_handlers_installed = true;
    return true;
}

/*
 * Put back the signal handlers that were active before the editor started.
 * This is the reverse of application_install_signal_handlers() and is safe
 * to call even if no handlers were installed.
 */
static void application_restore_signal_handlers(void)
{
    if (!signal_handlers_installed)
    {
        return;
    }

    (void)sigaction(SIGHUP, &previous_hup_action, NULL);
    (void)sigaction(SIGTERM, &previous_term_action, NULL);
    (void)sigaction(SIGINT, &previous_int_action, NULL);
    (void)sigaction(SIGWINCH, &previous_winch_action, NULL);
    signal_handlers_installed = false;
}

/*
 * Read a single byte from standard input, waiting at most
 * ESCAPE_TIMEOUT_MICROSECONDS for it to arrive.
 *
 * Used only while the Input Parser is inside an incomplete Escape sequence.
 * Arrow keys and friends arrive as several bytes, but a lone ESC key is
 * followed by nothing; the timeout lets us report it as an Escape event
 * instead of blocking forever.
 *
 * Parameters:
 *   byte - out-parameter: receives the byte read on success.
 *
 * Returns:
 *   1 if one byte was read, 0 if the wait timed out or the input stream
 *   closed, -1 on a read error. `byte` is written only when 1 is returned.
 */
static int read_escape_byte(unsigned char *byte)
{
    fd_set read_set;
    struct timeval timeout;
    int result;

    FD_ZERO(&read_set);
    FD_SET(STDIN_FILENO, &read_set);
    timeout.tv_sec = 0;
    timeout.tv_usec = ESCAPE_TIMEOUT_MICROSECONDS;

    do
    {
        result = select(STDIN_FILENO + 1, &read_set, NULL, NULL, &timeout);
    } while (result == -1 && errno == EINTR);

    if (result <= 0)
    {
        return result;
    }

    for (;;)
    {
        const ssize_t bytes_read = read(STDIN_FILENO, byte, sizeof(*byte));

        if (bytes_read == 1)
        {
            return 1;
        }
        if (bytes_read == 0)
        {
            return 0;
        }
        if (errno != EINTR)
        {
            return -1;
        }
    }
}

/*
 * Bring the cursor back into view (scrolling the Viewport if needed) and
 * redraw the whole screen.
 *
 * Called after every command that may have moved the cursor or changed the
 * document, so what is shown always matches the editor state.
 *
 * Parameters:
 *   renderer - the renderer that owns the off-screen frame buffer.
 *   state    - the current editor state to scroll and draw; not modified.
 *
 * Returns:
 *   true if the frame was drawn, false on NULL arguments or if scrolling or
 *   rendering failed.
 */
static bool redraw_after_action(Renderer *renderer, EditorState *state)
{
    if (renderer == NULL || state == NULL)
    {
        return false;
    }

    size_t cursor_visual_column;
    if (!renderer_cursor_visual_column(state, &cursor_visual_column) ||
        !viewport_ensure_cursor_visible(
            &state->viewport,
            state->cursor.row,
            cursor_visual_column,
            renderer->document_rows,
            renderer->screen_buffer.columns))
    {
        return false;
    }

    return renderer_render(renderer, state);
}

/*
 * Rebuild the Renderer for the latest terminal dimensions. A replacement is
 * allocated first, so allocation failure leaves the existing Renderer valid
 * for cleanup. The terminal dimensions are committed only after initialization
 * succeeds.
 *
 * Parameters:
 *   terminal - the terminal state whose dimensions are re-read.
 *   state    - the editor state to redraw after resizing.
 *   renderer - the renderer to replace; on success it holds the new frame
 *              buffer.
 *
 * Returns:
 *   true on success, false on NULL arguments, an unchanged or unusably small
 *   terminal size, allocation failure, or a redraw failure.
 */
static bool application_handle_resize(TerminalState *terminal, EditorState *state, Renderer *renderer)
{
    size_t rows;
    size_t columns;
    Renderer replacement;

    if (terminal == NULL || state == NULL || renderer == NULL ||
        !terminal_get_size(&rows, &columns))
    {
        return false;
    }

    if (rows == terminal->rows && columns == terminal->columns)
    {
        return true;
    }
    TEDIT_LOG_INFO("terminal", "resize requested: rows=%zu columns=%zu", rows, columns);

    /* The Renderer needs at least one document row and reserves the last
     * physical terminal column to avoid delayed-wrap behavior. */
    if (rows < 4 || columns < 2)
    {
        TEDIT_LOG_WARN("terminal", "rejected terminal size rows=%zu columns=%zu", rows, columns);
        return false;
    }

    if (!renderer_init(&replacement, rows, columns - 1))
    {
        TEDIT_LOG_ERROR("renderer", "failed to allocate replacement after terminal resize");
        return false;
    }

    Renderer previous = *renderer;
    *renderer = replacement;
    terminal->rows = rows;
    terminal->columns = columns;
    renderer_destroy(&previous);
    TEDIT_LOG_INFO("renderer", "renderer resized to rows=%zu columns=%zu", rows, columns - 1);

    if (!redraw_after_action(renderer, state))
    {
        TEDIT_LOG_ERROR("renderer", "redraw failed after terminal resize");
        return false;
    }
    return true;
}

/*
 * Run the editor's main loop until the user quits.
 *
 * Each iteration performs the whole pipeline: read one byte, let the Input
 * Parser turn bytes into a key event, translate that event into a command,
 * execute the command against the editor state, and redraw when anything
 * changed.
 *
 * Parameters:
 *   terminal  - the terminal state used to query dimensions on resize.
 *   state     - the editor state to operate on.
 *   renderer  - the renderer used to redraw after each command.
 *   save_path - path passed on the command line; used as the save target
 *               when the document has no file path of its own yet. May be
 *               NULL.
 *
 * Returns:
 *   true when the user quit normally, false on an unrecoverable error.
 */
static bool application_event_loop(TerminalState *terminal, EditorState *state, Renderer *renderer, const char *save_path)
{
    InputParser parser;
    input_parser_init(&parser);

    while (true)
    {
        if (termination_requested)
        {
            return true;
        }
        if (resize_requested)
        {
            resize_requested = 0;
            if (!application_handle_resize(terminal, state, renderer))
            {
                return false;
            }
        }

        unsigned char byte;
        int read_result;
        InputEvent event;
        bool should_quit;
        bool event_ready = false;

        /*
         * While the parser is in the middle of an Escape sequence, the next
         * byte must arrive quickly or not at all: read with a timeout so a
         * lone ESC press is handled promptly. Otherwise a plain blocking
         * read is enough.
         */
        if (input_parser_is_waiting(&parser))
        {
            read_result = read_escape_byte(&byte);

            if (read_result == 0)
            {
                if (!input_parser_timeout(&parser, &event))
                {
                    TEDIT_LOG_ERROR("input", "parser timeout handling failed");
                    return false;
                }

                if (event.type == INPUT_EVENT_NONE)
                {
                    continue;
                }

                event_ready = true;
                read_result = 1;
            }
        }
        else
        {
            const ssize_t bytes_read = read(STDIN_FILENO, &byte, sizeof(byte));

            if (bytes_read < 0)
            {
                if (errno == EINTR)
                {
                    continue;
                }
                TEDIT_LOG_ERROR("input", "terminal input read failed: errno=%d", errno);
                return false;
            }

            if (bytes_read == 0)
            {
                return true;
            }
            read_result = 1;
        }

        if (read_result < 0)
        {
            return false;
        }
        if (read_result == 0)
        {
            return true;
        }

        /*
         * Feed the byte into the parser; it returns true once a complete key
         * event is ready (or when a timeout already produced one above).
         */
        if (!event_ready && !input_parser_feed(&parser, byte, &event))
        {
            TEDIT_LOG_ERROR("input", "Input Parser rejected byte 0x%02X", (unsigned int)byte);
            return false;
        }

        if (event.type != INPUT_EVENT_NONE)
        {
            if (event.type == INPUT_EVENT_CHARACTER)
            {
                TEDIT_LOG_DEBUG("input", "recognized event=%s codepoint=U+%04X", application_input_event_name(event.type), (unsigned int)event.character);
            }
            else
            {
                TEDIT_LOG_DEBUG("input", "recognized event=%s", application_input_event_name(event.type));
            }
        }

        EditorCommand command;
        bool redraw;

        /* Pipeline step by step: event -> command -> state change -> redraw. */
        if (!input_engine_translate(&event, &command))
        {
            TEDIT_LOG_ERROR("input", "Input Engine failed to translate event=%s",
                            application_input_event_name(event.type));
            return false;
        }

        if (command.type != EDITOR_COMMAND_NONE)
        {
            TEDIT_LOG_DEBUG("command", "executing command=%s", application_command_name(command.type));
        }
        command.page_rows = renderer->document_rows;

        if (!command_engine_execute(state, &command, save_path, &redraw, &should_quit))
        {
            TEDIT_LOG_ERROR("command", "Command Engine failed for command=%s", application_command_name(command.type));
            return false;
        }
        if (redraw && !redraw_after_action(renderer, state))
        {
            TEDIT_LOG_ERROR("renderer", "redraw failed after command=%s", application_command_name(command.type));
            return false;
        }
        if (should_quit)
        {
            return true;
        }
    }
}

/*
 * Program entry point.
 *
 * Sets up the terminal (raw mode + alternate screen), initializes the editor
 * state, optionally loads the file named on the command line, runs the event
 * loop, and then restores everything - including after errors - before
 * returning to the shell.
 *
 * Parameters:
 *   argc - number of command-line arguments.
 *   argv - argument vector; argv[1] may name a file to open.
 *
 * Returns:
 *   0 on a clean exit, 1 on any startup or runtime error.
 */
int main(int argc, char *argv[])
{
    (void)setlocale(LC_CTYPE, "");

#ifdef TEDIT_ENABLE_LOGGING
#ifdef NDEBUG
    (void)logger_init("log.txt", LOGGER_LEVEL_INFO);
#else
    (void)logger_init("log.txt", LOGGER_LEVEL_DEBUG);
#endif
    TEDIT_LOG_INFO("lifecycle", "editor process started");
#endif

    if (argc > 2)
    {
        TEDIT_LOG_ERROR("lifecycle", "invalid command-line argument count: %d", argc);
        fprintf(stderr, "Usage: %s [filename]\n", argv[0]);
#ifdef TEDIT_ENABLE_LOGGING
        logger_shutdown();
#endif
        return 1;
    }

    const char *startup_path = argc == 2 ? argv[1] : NULL;
    bool startup_load_failed = false;
    TerminalState terminal;
    EditorState state;
    Renderer renderer;
    bool state_initialized = false;
    bool renderer_initialized = false;
    int exit_status = 1;

    if (!application_install_signal_handlers())
    {
        TEDIT_LOG_ERROR("lifecycle", "failed to install signal handlers");
        goto early_cleanup;
    }
    if (!terminal_init(&terminal))
    {
        TEDIT_LOG_ERROR("terminal", "terminal initialization failed");
        application_restore_signal_handlers();
        goto early_cleanup;
    }
    if (!terminal_enable_raw_mode(&terminal))
    {
        TEDIT_LOG_ERROR("terminal", "failed to enable raw mode");
        terminal_restore(&terminal);
        application_restore_signal_handlers();
        goto early_cleanup;
    }
    if (!terminal_enter_alternate_screen(&terminal))
    {
        TEDIT_LOG_ERROR("terminal", "failed to enter alternate screen");
        terminal_restore(&terminal);
        application_restore_signal_handlers();
        goto early_cleanup;
    }
    if (!editor_state_init(&state))
    {
        TEDIT_LOG_ERROR("state", "Editor State initialization failed");
        goto cleanup;
    }
    TEDIT_LOG_INFO("state", "Editor State initialized");
    state_initialized = true;

    if (startup_path != NULL && !command_engine_load_file(&state, startup_path))
    {
        startup_load_failed = true;
        TEDIT_LOG_ERROR("file", "startup file load failed: path=\"%s\"", startup_path);
        goto cleanup;
    }
    if (terminal.columns < 2 || !renderer_init(&renderer, terminal.rows, terminal.columns - 1))
    {
        TEDIT_LOG_ERROR("renderer", "initialization failed for terminal rows=%zu columns=%zu", terminal.rows, terminal.columns);
        goto cleanup;
    }
    TEDIT_LOG_INFO("renderer", "initialized rows=%zu columns=%zu", terminal.rows, terminal.columns - 1);
    renderer_initialized = true;

    if (!redraw_after_action(&renderer, &state))
    {
        TEDIT_LOG_ERROR("renderer", "initial frame rendering failed");
        goto cleanup;
    }
    if (!application_event_loop(&terminal, &state, &renderer, startup_path))
    {
        TEDIT_LOG_ERROR("lifecycle", "event loop terminated after an unrecoverable error");
        goto cleanup;
    }

    exit_status = 0;

cleanup:
    if (renderer_initialized)
    {
        renderer_destroy(&renderer);
        TEDIT_LOG_DEBUG("renderer", "Renderer resources released");
    }
    if (state_initialized)
    {
        editor_state_destroy(&state);
        TEDIT_LOG_DEBUG("state", "Editor State resources released");
    }
    terminal_restore(&terminal);
    terminal_leave_alternate_screen(&terminal);
    application_restore_signal_handlers();

    if (startup_load_failed)
    {
        fprintf(stderr, "tedit: unable to open or decode '%s'.\n", startup_path);
    }

    TEDIT_LOG_INFO("lifecycle", "editor process stopped with status=%d", exit_status);
#ifdef TEDIT_ENABLE_LOGGING
    logger_shutdown();
#endif
    return exit_status;

early_cleanup:
    TEDIT_LOG_INFO("lifecycle", "editor process stopped during startup with status=1");
#ifdef TEDIT_ENABLE_LOGGING
    logger_shutdown();
#endif
    return 1;
}
```

```c
static void write_all(int descriptor, const unsigned char *data, size_t length)
{
    size_t offset = 0;

    while (offset < length)
    {
        ssize_t written = write(
            descriptor,
            data + offset,
            length - offset);

        assert(written > 0);
        offset += (size_t)written;
    }
}
```

```c
static void write_all(int descriptor, const unsigned char *data, size_t length)
{
    size_t offset = 0;

    while (offset < length)
    {
        ssize_t written = write(descriptor, data + offset, length - offset);
        assert(written > 0);
        offset += (size_t)written;
    }
}

static size_t read_file(const char *path, unsigned char *buffer, size_t capacity)
{
    FILE *file = fopen(path, "rb");
    size_t length;

    assert(file != NULL);
    length = fread(buffer, 1, capacity, file);
    assert(!ferror(file));
    assert(feof(file) || length == capacity);
    assert(fclose(file) == 0);
    return length;
}
```
