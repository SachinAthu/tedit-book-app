---
title: 52.7 Filename Entry and Save As
description: "Adds minimal filename entry and Save As, allowing unnamed documents to be saved while preserving state on cancellation or failure."
---

## Implementation goal

Add a minimal filename-entry flow using the existing input pipeline. Ctrl-S continues saving to the associated file. When no file path is associated, Ctrl-S opens filename entry. Printable characters append to the path, Backspace removes the previous UTF-8 code point, Enter confirms the save, and Escape cancels.

## Invariants

- Cancellation does not change the current file association, Document, or Dirty Flag.
- A failed save does not change the current file association, Document, or Dirty Flag. The prompt remains active so the path can be corrected or cancelled.
- A successful save updates the file association and clears the Dirty Flag only after the File Manager reports success.
- The Command Engine does not perform terminal I/O. The Renderer remains the only component that draws the prompt.

## Changed files

- `src/state/editor_state.h` and `.c`: filename-entry state and `editor_state_save_as()`.
- `src/input/input_parser.h` and `.c`: standalone Escape event on timeout.
- `src/input/input_engine.c`: Escape-to-cancel mapping.
- `src/command/command_engine.h` and `.c`: filename-entry state machine and Save As execution.
- `src/renderer/renderer.c`: prompt display and cursor placement.
- `src/main.c`: dispatches the Escape timeout event through the normal input pipeline.
- `tests/test_input_parser.c`, `test_input_engine.c`, `test_command_engine.c`, `test_input_pipeline.c`, and `test_file_save.c`: regression coverage.

```c
/* src/state/editor_state.c */

#include "editor_state.h"
#include "file/file.h"

#include <string.h>
#include <stdlib.h>

bool editor_state_init(EditorState *state)
{
    if (state == NULL)
    {
        return false;
    }

    /* Zero every field first so partial failures leave a clean state. */
    memset(state, 0, sizeof(*state));

    if (!document_init(&state->document))
    {
        return false;
    }

    cursor_init(&state->cursor);
    viewport_init(&state->viewport);
    state->is_dirty = false;
    state->file_path = NULL;
    state->message[0] = '\0';
    state->filename_entry_active = false;
    state->filename_entry[0] = '\0';
    state->filename_entry_length = 0;

    return true;
}

bool editor_state_load_file(EditorState *state, const char *path)
{
    if (state == NULL || path == NULL)
    {
        return false;
    }

    char *new_path;
    size_t path_length = strlen(path);

    if (path_length == SIZE_MAX)
    {
        return false;
    }

    new_path = malloc(path_length + 1);
    if (new_path == NULL)
    {
        return false;
    }
    memcpy(new_path, path, path_length + 1);

    /* Do not change the current state or path unless loading succeeds. */
    if (file_load_document(path, &state->document) != FILE_LOAD_OK)
    {
        free(new_path);
        return false;
    }

    free(state->file_path);
    state->file_path = new_path;
    cursor_init(&state->cursor);
    viewport_init(&state->viewport);
    state->is_dirty = false;
    state->filename_entry_active = false;
    state->filename_entry[0] = '\0';
    state->filename_entry_length = 0;

    return true;
}

bool editor_state_save_file(EditorState *state, const char *path)
{
    if (state == NULL || path == NULL)
    {
        return false;
    }

    if (file_save_document(path, &state->document) != FILE_SAVE_OK)
    {
        return false;
    }

    state->is_dirty = false;
    return true;
}

bool editor_state_save_as(EditorState *state, const char *path)
{
    char *new_path;
    size_t path_length;

    if (state == NULL || path == NULL || path[0] == '\0')
    {
        return false;
    }

    path_length = strlen(path);
    if (path_length == SIZE_MAX)
    {
        return false;
    }

    new_path = malloc(path_length + 1);
    if (new_path == NULL)
    {
        return false;
    }
    memcpy(new_path, path, path_length + 1);

    if (file_save_document(path, &state->document) != FILE_SAVE_OK)
    {
        free(new_path);
        return false;
    }

    free(state->file_path);
    state->file_path = new_path;
    state->is_dirty = false;
    return true;
}

bool editor_state_set_message(EditorState *state, const char *message)
{
    size_t length;

    if (state == NULL || message == NULL)
    {
        return false;
    }

    length = strlen(message);
    if (length >= sizeof(state->message))
    {
        length = sizeof(state->message) - 1;
    }

    memmove(state->message, message, length);
    state->message[length] = '\0';
    return true;
}

void editor_state_destroy(EditorState *state)
{
    if (state == NULL)
    {
        return;
    }

    document_destroy(&state->document);
    free(state->file_path);
    state->file_path = NULL;
    cursor_init(&state->cursor);
    viewport_init(&state->viewport);
    state->is_dirty = false;
    state->message[0] = '\0';
    state->filename_entry_active = false;
    state->filename_entry[0] = '\0';
    state->filename_entry_length = 0;
}
```

```c
/* src/input/input_parser.c */

#include "input/input_parser.h"

#include <stddef.h>

#define INPUT_CONTROL_Q 0x11
#define INPUT_CONTROL_S 0x13
#define INPUT_BACKSPACE 0x7F
#define INPUT_BACKSPACE_ALT 0x08
#define INPUT_ESCAPE 0x1B
#define INPUT_ENTER 0x0D
#define INPUT_LINE_FEED 0x0A
#define UTF8_CONTINUATION_MASK 0xC0
#define UTF8_CONTINUATION_PREFIX 0x80
#define UTF8_TWO_BYTE_PREFIX 0xC0
#define UTF8_THREE_BYTE_PREFIX 0xE0
#define UTF8_FOUR_BYTE_PREFIX 0xF0

void input_parser_init(InputParser *parser)
{
    if (parser == NULL)
    {
        return;
    }

    parser->state = INPUT_PARSER_NORMAL;
    parser->utf8_code_point = 0;
    parser->utf8_minimum = 0;
    parser->utf8_remaining = 0;
}

static void input_event_clear(InputEvent *event)
{
    event->type = INPUT_EVENT_NONE;
    event->character = 0;
}

static bool input_parser_feed_utf8(InputParser *parser, unsigned char byte, InputEvent *event)
{
    if (parser->utf8_remaining > 0)
    {
        if ((byte & UTF8_CONTINUATION_MASK) != UTF8_CONTINUATION_PREFIX)
        {
            parser->utf8_remaining = 0;
            parser->utf8_code_point = 0;
            parser->utf8_minimum = 0;
            return input_parser_feed(parser, byte, event);
        }

        parser->utf8_code_point = (parser->utf8_code_point << 6) | (uint32_t)(byte & 0x3F);
        parser->utf8_remaining--;

        if (parser->utf8_remaining == 0)
        {
            const uint32_t code_point = parser->utf8_code_point;
            const bool valid = code_point >= parser->utf8_minimum &&
                               code_point <= 0x10FFFF &&
                               !(code_point >= 0xD800 && code_point <= 0xDFFF);
            parser->utf8_code_point = 0;
            parser->utf8_minimum = 0;

            if (valid)
            {
                event->type = INPUT_EVENT_CHARACTER;
                event->character = code_point;
            }
        }

        return true;
    }

    if (byte >= 0xC2 && byte <= 0xDF)
    {
        parser->utf8_code_point = (uint32_t)(byte & 0x1F);
        parser->utf8_minimum = 0x80;
        parser->utf8_remaining = 1;
    }
    else if (byte >= 0xE0 && byte <= 0xEF)
    {
        parser->utf8_code_point = (uint32_t)(byte & 0x0F);
        parser->utf8_minimum = 0x800;
        parser->utf8_remaining = 2;
    }
    else if (byte >= 0xF0 && byte <= 0xF4)
    {
        parser->utf8_code_point = (uint32_t)(byte & 0x07);
        parser->utf8_minimum = 0x10000;
        parser->utf8_remaining = 3;
    }

    return true;
}

bool input_parser_feed(InputParser *parser, unsigned char byte, InputEvent *event)
{
    if (parser == NULL || event == NULL)
    {
        return false;
    }

    input_event_clear(event);

    if (parser->utf8_remaining > 0)
    {
        return input_parser_feed_utf8(parser, byte, event);
    }

    if (parser->state == INPUT_PARSER_NORMAL)
    {
        if (byte == INPUT_ESCAPE)
        {
            parser->state = INPUT_PARSER_ESCAPE;
        }
        else if (byte == INPUT_CONTROL_Q)
        {
            event->type = INPUT_EVENT_CTRL_Q;
        }
        else if (byte == INPUT_CONTROL_S)
        {
            event->type = INPUT_EVENT_CTRL_S;
        }
        else if (byte == INPUT_BACKSPACE || byte == INPUT_BACKSPACE_ALT)
        {
            event->type = INPUT_EVENT_BACKSPACE;
        }
        else if (byte == INPUT_ENTER || byte == INPUT_LINE_FEED)
        {
            event->type = INPUT_EVENT_ENTER;
        }
        else if (byte >= 0x20 && byte <= 0x7E)
        {
            event->type = INPUT_EVENT_CHARACTER;
            event->character = byte;
        }
        else if (byte >= 0x80)
        {
            return input_parser_feed_utf8(parser, byte, event);
        }

        return true;
    }

    if (parser->state == INPUT_PARSER_ESCAPE)
    {
        if (byte == '[')
        {
            parser->state = INPUT_PARSER_CSI;
        }
        else if (byte == INPUT_ESCAPE)
        {
            parser->state = INPUT_PARSER_ESCAPE;
        }
        else
        {
            parser->state = INPUT_PARSER_NORMAL;
        }

        return true;
    }

    parser->state = INPUT_PARSER_NORMAL;

    if (byte == 'D')
    {
        event->type = INPUT_EVENT_ARROW_LEFT;
    }
    else if (byte == 'C')
    {
        event->type = INPUT_EVENT_ARROW_RIGHT;
    }
    else if (byte == 'A')
    {
        event->type = INPUT_EVENT_ARROW_UP;
    }
    else if (byte == 'B')
    {
        event->type = INPUT_EVENT_ARROW_DOWN;
    }

    return true;
}

bool input_parser_timeout(InputParser *parser, InputEvent *event)
{
    if (parser == NULL || event == NULL)
    {
        return false;
    }

    const InputParserState previous_state = parser->state;

    parser->state = INPUT_PARSER_NORMAL;
    parser->utf8_code_point = 0;
    parser->utf8_minimum = 0;
    parser->utf8_remaining = 0;
    input_event_clear(event);
    if (previous_state == INPUT_PARSER_ESCAPE)
    {
        event->type = INPUT_EVENT_ESCAPE;
    }
    return true;
}

bool input_parser_is_waiting(const InputParser *parser)
{
    return parser != NULL && parser->state != INPUT_PARSER_NORMAL;
}
```

```c
#include "input/input_engine.h"

static void command_set(EditorCommand *command, EditorCommandType type, uint32_t character)
{
    command->type = type;
    command->character = character;
}

bool input_engine_translate(const InputEvent *event, EditorCommand *command)
{
    if (event == NULL || command == NULL)
    {
        return false;
    }

    command_set(command, EDITOR_COMMAND_NONE, 0);

    switch (event->type)
    {
        case INPUT_EVENT_NONE:
            break;
        case INPUT_EVENT_CHARACTER:
            command_set(command, EDITOR_COMMAND_INSERT_CHARACTER, event->character);
            break;
        case INPUT_EVENT_CTRL_Q:
            command_set(command, EDITOR_COMMAND_QUIT, 0);
            break;
        case INPUT_EVENT_CTRL_S:
            command_set(command, EDITOR_COMMAND_SAVE, 0);
            break;
        case INPUT_EVENT_BACKSPACE:
            command_set(command, EDITOR_COMMAND_BACKSPACE, 0);
            break;
        case INPUT_EVENT_ENTER:
            command_set(command, EDITOR_COMMAND_INSERT_NEWLINE, 0);
            break;
        case INPUT_EVENT_ESCAPE:
            command_set(command, EDITOR_COMMAND_CANCEL, 0);
            break;
        case INPUT_EVENT_ARROW_LEFT:
            command_set(command, EDITOR_COMMAND_MOVE_LEFT, 0);
            break;
        case INPUT_EVENT_ARROW_RIGHT:
            command_set(command, EDITOR_COMMAND_MOVE_RIGHT, 0);
            break;
        case INPUT_EVENT_ARROW_UP:
            command_set(command, EDITOR_COMMAND_MOVE_UP, 0);
            break;
        case INPUT_EVENT_ARROW_DOWN:
            command_set(command, EDITOR_COMMAND_MOVE_DOWN, 0);
            break;
        default:
            return false;
    }

    return true;
}
```

```c
/* src/command/command_engine.c */

#include "command/command_engine.h"

#include "cursor/cursor.h"
#include "document/document.h"
#include "state/editor_state.h"

static bool command_backspace(EditorState *state, bool *changed)
{
    if (state->cursor.column > 0)
    {
        const size_t delete_column = state->cursor.column - 1;
        if (!document_delete_character(&state->document, state->cursor.row, delete_column))
        {
            return true;
        }
        if (!cursor_set_position(&state->cursor, &state->document,
                                 state->cursor.row, delete_column))
        {
            return false;
        }
    }
    else if (state->cursor.row > 0)
    {
        const size_t previous_row = state->cursor.row - 1;
        const DocumentLine *previous_line = document_get_line(&state->document, previous_row);
        if (previous_line == NULL)
        {
            return true;
        }

        const size_t previous_length = previous_line->length;
        if (!document_merge_lines(&state->document, previous_row))
        {
            return true;
        }
        if (!cursor_set_position(&state->cursor, &state->document,
                                 previous_row, previous_length))
        {
            return false;
        }
    }
    else
    {
        return true;
    }

    state->is_dirty = true;
    *changed = true;
    return true;
}

static size_t encode_utf8(uint32_t character, unsigned char bytes[4])
{
    if (character <= 0x7F)
    {
        bytes[0] = (unsigned char)character;
        return 1;
    }
    if (character <= 0x7FF)
    {
        bytes[0] = (unsigned char)(0xC0 | (character >> 6));
        bytes[1] = (unsigned char)(0x80 | (character & 0x3F));
        return 2;
    }
    if (character <= 0xFFFF && !(character >= 0xD800 && character <= 0xDFFF))
    {
        bytes[0] = (unsigned char)(0xE0 | (character >> 12));
        bytes[1] = (unsigned char)(0x80 | ((character >> 6) & 0x3F));
        bytes[2] = (unsigned char)(0x80 | (character & 0x3F));
        return 3;
    }
    if (character <= 0x10FFFF)
    {
        bytes[0] = (unsigned char)(0xF0 | (character >> 18));
        bytes[1] = (unsigned char)(0x80 | ((character >> 12) & 0x3F));
        bytes[2] = (unsigned char)(0x80 | ((character >> 6) & 0x3F));
        bytes[3] = (unsigned char)(0x80 | (character & 0x3F));
        return 4;
    }
    return 0;
}

static bool filename_entry_execute(EditorState *state,
                                   const EditorCommand *command,
                                   bool *redraw)
{
    if (command->type == EDITOR_COMMAND_CANCEL)
    {
        state->filename_entry_active = false;
        state->filename_entry_length = 0;
        state->filename_entry[0] = '\0';
        (void)editor_state_set_message(state, "Save As cancelled.");
        *redraw = true;
        return true;
    }

    if (command->type == EDITOR_COMMAND_BACKSPACE)
    {
        if (state->filename_entry_length > 0)
        {
            --state->filename_entry_length;
            while (state->filename_entry_length > 0 &&
                   (((unsigned char)state->filename_entry[state->filename_entry_length] & 0xC0) == 0x80))
            {
                --state->filename_entry_length;
            }
            state->filename_entry[state->filename_entry_length] = '\0';
            *redraw = true;
        }
        return true;
    }

    if (command->type == EDITOR_COMMAND_INSERT_CHARACTER)
    {
        unsigned char encoded[4];
        const size_t encoded_length = encode_utf8(command->character, encoded);

        if (encoded_length == 0)
        {
            return true;
        }
        if (encoded_length > EDITOR_FILENAME_CAPACITY - 1 - state->filename_entry_length)
        {
            (void)editor_state_set_message(state, "Filename is too long.");
            *redraw = true;
            return true;
        }

        for (size_t index = 0; index < encoded_length; ++index)
        {
            state->filename_entry[state->filename_entry_length++] = (char)encoded[index];
        }
        state->filename_entry[state->filename_entry_length] = '\0';
        *redraw = true;
        return true;
    }

    if (command->type == EDITOR_COMMAND_INSERT_NEWLINE)
    {
        if (state->filename_entry_length == 0)
        {
            (void)editor_state_set_message(state, "Enter a file name.");
            *redraw = true;
            return true;
        }

        if (editor_state_save_as(state, state->filename_entry))
        {
            state->filename_entry_active = false;
            state->filename_entry_length = 0;
            state->filename_entry[0] = '\0';
            (void)editor_state_set_message(state, "Saved successfully.");
        }
        else
        {
            (void)editor_state_set_message(
                state, "Save failed; check path and try again, or Esc to cancel.");
        }
        *redraw = true;
        return true;
    }

    /* Ignore navigation and other commands while filename entry is active. */
    return true;
}

bool command_engine_execute(EditorState *state,
                            const EditorCommand *command,
                            const char *save_path,
                            bool *redraw,
                            bool *should_quit)
{
    bool changed = false;

    if (state == NULL || command == NULL || redraw == NULL || should_quit == NULL)
    {
        return false;
    }

    *redraw = false;
    *should_quit = false;

    if (state->filename_entry_active)
    {
        return filename_entry_execute(state, command, redraw);
    }

    switch (command->type)
    {
        case EDITOR_COMMAND_NONE:
            break;
        case EDITOR_COMMAND_QUIT:
            *should_quit = true;
            break;
        case EDITOR_COMMAND_SAVE:
        {
            const char *current_path = state->file_path != NULL ? state->file_path : save_path;

            if (current_path == NULL)
            {
                state->filename_entry_active = true;
                state->filename_entry_length = 0;
                state->filename_entry[0] = '\0';
                (void)editor_state_set_message(state, "Enter: save  Esc: cancel");
            }
            else if (editor_state_save_file(state, current_path))
            {
                (void)editor_state_set_message(state, "Saved successfully.");
            }
            else
            {
                (void)editor_state_set_message(state, "Save failed; document remains modified.");
            }
            *redraw = true;
            break;
        }
        case EDITOR_COMMAND_CANCEL:
            break;
        case EDITOR_COMMAND_INSERT_CHARACTER:
            if (!document_insert_character(&state->document, state->cursor.row,
                                           state->cursor.column, command->character))
            {
                return true;
            }
            if (!cursor_advance(&state->cursor, &state->document))
            {
                return false;
            }
            state->is_dirty = true;
            changed = true;
            break;
        case EDITOR_COMMAND_BACKSPACE:
            if (!command_backspace(state, &changed))
            {
                return false;
            }
            break;
        case EDITOR_COMMAND_INSERT_NEWLINE:
            if (!document_split_line(&state->document, state->cursor.row, state->cursor.column))
            {
                return true;
            }
            if (!cursor_set_position(&state->cursor, &state->document,
                                     state->cursor.row + 1, 0))
            {
                return false;
            }
            state->is_dirty = true;
            changed = true;
            break;
        case EDITOR_COMMAND_MOVE_LEFT:
            changed = cursor_move_left(&state->cursor);
            break;
        case EDITOR_COMMAND_MOVE_RIGHT:
            changed = cursor_move_right(&state->cursor, &state->document);
            break;
        case EDITOR_COMMAND_MOVE_UP:
            changed = cursor_move_up(&state->cursor, &state->document);
            break;
        case EDITOR_COMMAND_MOVE_DOWN:
            changed = cursor_move_down(&state->cursor, &state->document);
            break;
        default:
            return false;
    }

    *redraw = *redraw || changed;
    return true;
}
```

```c
/* src/renderer/renderer.c */

#include "renderer.h"

#include <errno.h>

/* Fill an in-memory ScreenBuffer with the visible part of the document. */
static bool renderer_build_frame(Renderer *renderer, const EditorState *state);

static bool renderer_draw_status_bar(Renderer *renderer, const EditorState *state);

static bool renderer_draw_message_bar(Renderer *renderer, const EditorState *state);

/* Send a finished ScreenBuffer to the terminal. */
static bool renderer_write_frame(Renderer *renderer);

/* Grow the output buffer so it can hold at least 'required' bytes. */
static bool renderer_ensure_output_capacity(Renderer *renderer, size_t required);

/* Write all bytes to the terminal, retrying if the write is interrupted. */
static bool renderer_write_all(const char *data, size_t length);

/* Move the terminal's visible cursor to a given row and column. */
static bool renderer_write_cursor_position(size_t row, size_t column);

static size_t renderer_utf8_length(uint32_t code_point);

static size_t renderer_encode_utf8(uint32_t code_point, char output[4]);

bool renderer_init(Renderer *renderer, size_t rows, size_t columns)
{
    if (renderer == NULL)
    {
        return false;
    }

    renderer->screen_buffer.cells = NULL;
    renderer->screen_buffer.rows = 0;
    renderer->screen_buffer.columns = 0;
    renderer->document_rows = 0;
    renderer->output_buffer = NULL;
    renderer->output_capacity = 0;

    if (rows < 3 || columns == 0 ||
        !screen_buffer_init(&renderer->screen_buffer, rows, columns))
    {
        return false;
    }

    renderer->document_rows = rows - 2;
    return true;
}

void renderer_destroy(Renderer *renderer)
{
    if (renderer == NULL)
    {
        return;
    }

    screen_buffer_destroy(&renderer->screen_buffer);

    free(renderer->output_buffer);
    renderer->output_buffer = NULL;
    renderer->output_capacity = 0;
}

bool renderer_render(Renderer *renderer, const EditorState *state)
{
    bool success;

    if (renderer == NULL || state == NULL || renderer->screen_buffer.cells == NULL)
    {
        return false;
    }
    if (!state->filename_entry_active &&
        (state->cursor.row < state->viewport.viewport_row ||
         state->cursor.column < state->viewport.viewport_column ||
         state->cursor.row - state->viewport.viewport_row >= renderer->document_rows ||
         state->cursor.column - state->viewport.viewport_column >= renderer->screen_buffer.columns))
    {
        return false;
    }

    if (!renderer_build_frame(renderer, state))
    {
        return false;
    }

    /*
     * Hide the terminal text cursor while drawing.
     */
    if (!renderer_write_all("\x1b[?25l", 6))
    {
        return false;
    }

    success = renderer_write_frame(renderer);

    if (success)
    {
        if (state->filename_entry_active)
        {
            const size_t prompt_length = sizeof("Save as: ") - 1;
            size_t entry_column = prompt_length + state->filename_entry_length;
            if (entry_column >= renderer->screen_buffer.columns)
            {
                entry_column = renderer->screen_buffer.columns - 1;
            }
            success = renderer_write_cursor_position(renderer->document_rows, entry_column);
        }
        else
        {
            success = renderer_write_cursor_position(
                state->cursor.row - state->viewport.viewport_row,
                state->cursor.column - state->viewport.viewport_column);
        }
    }

    /*
     * Always attempt to restore cursor visibility,
     * even if frame rendering fails.
     */
    if (!renderer_write_all("\x1b[?25h", 6))
    {
        success = false;
    }

    return success;
}

static bool renderer_build_frame(Renderer *renderer, const EditorState *state)
{
    if (renderer == NULL || state == NULL)
    {
        return false;
    }

    if (renderer->screen_buffer.cells == NULL)
    {
        return false;
    }

    /*
     * Start from a clean slate: every cell becomes a blank space. A
     * zero-line Document (the representation of an empty file) therefore
     * renders as a blank document area without requiring a synthetic line.
     */
    screen_buffer_clear(&renderer->screen_buffer, (uint32_t)' ');

    /* Map each screen row to its corresponding logical document row. */
    for (size_t screen_row = 0; screen_row < renderer->document_rows; screen_row++)
    {
        const size_t document_row = state->viewport.viewport_row + screen_row;
        const DocumentLine *line;

        if (document_row >= state->document.line_count)
        {
            break;
        }

        line = &state->document.lines[document_row];

        /* Horizontal scrolling maps screen columns to document columns. */
        for (size_t screen_column = 0; screen_column < renderer->screen_buffer.columns; screen_column++)
        {
            const size_t document_column = state->viewport.viewport_column + screen_column;

            if (document_column >= line->length)
            {
                break;
            }

            if (!screen_buffer_set(&renderer->screen_buffer, screen_row, screen_column, line->data[document_column]))
            {
                return false;
            }
        }
    }

    return renderer_draw_status_bar(renderer, state) &&
           renderer_draw_message_bar(renderer, state);
}

/* Draw filename, modified state, and one-based cursor position in the Status Bar row. */
static bool renderer_draw_status_bar(Renderer *renderer, const EditorState *state)
{
    char right[96];
    const char *path;
    const char *suffix;
    size_t path_length;
    size_t suffix_length;
    int right_length;
    size_t status_row;
    size_t left_limit;
    size_t path_limit;
    size_t path_position = 0;
    size_t right_position;

    if (renderer == NULL || state == NULL || renderer->document_rows == 0)
    {
        return false;
    }

    path = state->file_path != NULL ? state->file_path : "[No Name]";
    path_length = strlen(path);
    suffix = state->is_dirty ? " [+]" : "";
    suffix_length = strlen(suffix);
    right_length = snprintf(right, sizeof(right), "Ln %zu, Col %zu",
                            state->cursor.row + 1, state->cursor.column + 1);
    if (right_length < 0 || (size_t)right_length >= sizeof(right))
    {
        return false;
    }

    status_row = renderer->document_rows;
    left_limit = renderer->screen_buffer.columns > (size_t)right_length + 2
                     ? renderer->screen_buffer.columns - (size_t)right_length - 2
                     : 0;
    path_limit = left_limit > suffix_length ? left_limit - suffix_length : 0;

    for (size_t column = 0; column < renderer->screen_buffer.columns; ++column)
    {
        if (!screen_buffer_set(&renderer->screen_buffer, status_row, column, (uint32_t)' '))
        {
            return false;
        }
    }

    if (state->filename_entry_active)
    {
        static const char prompt[] = "Save as: ";
        size_t column = 0;

        while (column < sizeof(prompt) - 1 && column < renderer->screen_buffer.columns)
        {
            if (!screen_buffer_set(&renderer->screen_buffer, status_row, column,
                                   (uint32_t)(unsigned char)prompt[column]))
            {
                return false;
            }
            ++column;
        }

        for (size_t index = 0; index < state->filename_entry_length &&
                               column < renderer->screen_buffer.columns; ++index, ++column)
        {
            const unsigned char byte = (unsigned char)state->filename_entry[index];
            const uint32_t code_point = byte >= 0x20 && byte < 0x7F
                                            ? byte
                                            : (byte >= 0x80 ? (uint32_t)'?' : (uint32_t)' ');
            if (!screen_buffer_set(&renderer->screen_buffer, status_row, column, code_point))
            {
                return false;
            }
        }
        return true;
    }

    /* Render a bounded path prefix; arbitrary filesystem bytes stay printable. */
    while (path_position < path_length && path_position < path_limit)
    {
        unsigned char byte = (unsigned char)path[path_position];
        uint32_t code_point = byte >= 0x20 && byte < 0x7F
                                  ? byte
                                  : (byte >= 0x80 ? (uint32_t)'?' : (uint32_t)' ');
        if (!screen_buffer_set(&renderer->screen_buffer, status_row, path_position, code_point))
        {
            return false;
        }
        ++path_position;
    }

    for (size_t index = 0; index < suffix_length && path_position + index < left_limit; ++index)
    {
        if (!screen_buffer_set(&renderer->screen_buffer, status_row, path_position + index,
                               (uint32_t)(unsigned char)suffix[index]))
        {
            return false;
        }
    }

    if ((size_t)right_length < renderer->screen_buffer.columns)
    {
        right_position = renderer->screen_buffer.columns - (size_t)right_length;
        for (size_t index = 0; index < (size_t)right_length; ++index)
        {
            if (!screen_buffer_set(&renderer->screen_buffer, status_row, right_position + index,
                                   (uint32_t)(unsigned char)right[index]))
            {
                return false;
            }
        }
    }

    return true;
}

/* Draw the current message on the final row, clipping it to screen width. */
static bool renderer_draw_message_bar(Renderer *renderer, const EditorState *state)
{
    size_t message_row;
    size_t message_length;
    size_t visible_length;

    if (renderer == NULL || state == NULL || renderer->screen_buffer.cells == NULL ||
        renderer->screen_buffer.rows < 3)
    {
        return false;
    }

    message_row = renderer->screen_buffer.rows - 1;
    message_length = strlen(state->message);
    visible_length = message_length < renderer->screen_buffer.columns
                         ? message_length
                         : renderer->screen_buffer.columns;

    for (size_t column = 0; column < renderer->screen_buffer.columns; ++column)
    {
        uint32_t code_point = (uint32_t)' ';

        if (column < visible_length)
        {
            unsigned char byte = (unsigned char)state->message[column];
            code_point = byte >= 0x20 && byte < 0x7F
                             ? (uint32_t)byte
                             : (byte >= 0x80 ? (uint32_t)'?' : (uint32_t)' ');
        }

        if (!screen_buffer_set(&renderer->screen_buffer, message_row, column, code_point))
        {
            return false;
        }
    }

    return true;
}

static bool renderer_ensure_output_capacity(Renderer *renderer, size_t required)
{
    if (renderer == NULL)
    {
        return false;
    }

    if (required <= renderer->output_capacity)
    {
        return true;
    }

    char *larger = realloc(renderer->output_buffer, required);
    if (larger == NULL)
    {
        return false;
    }

    renderer->output_buffer = larger;
    renderer->output_capacity = required;

    return true;
}

static bool renderer_write_frame(Renderer *renderer)
{
    const ScreenBuffer *buffer;
    char sequence[64];

    if (renderer == NULL)
    {
        return false;
    }

    buffer = &renderer->screen_buffer;

    if (buffer->cells == NULL)
    {
        return false;
    }

    /*
     * First pass: measure the ANSI cursor-position sequences and
     * UTF-8-encoded cell contents.
     */
    size_t frame_length = 0;

    for (size_t row = 0; row < buffer->rows; row++)
    {
        int sequence_length = snprintf(sequence, sizeof(sequence), "\x1b[%zu;1H", row + 1);

        if (sequence_length < 0 || (size_t)sequence_length >= sizeof(sequence))
        {
            return false;
        }

        if (frame_length > SIZE_MAX - (size_t)sequence_length)
        {
            return false;
        }

        frame_length += (size_t)sequence_length;

        for (size_t column = 0; column < buffer->columns; ++column)
        {
            const ScreenCell *cell = screen_buffer_get(buffer, row, column);

            if (cell == NULL)
            {
                return false;
            }

            size_t encoded_length = renderer_utf8_length(cell->code_point);

            if (encoded_length == 0 || frame_length > SIZE_MAX - encoded_length)
            {
                return false;
            }

            frame_length += encoded_length;
        }
    }

    if (!renderer_ensure_output_capacity(renderer, frame_length))
    {
        return false;
    }

    /*
     * Second pass: construct the complete terminal output.
     */
    size_t position = 0;

    for (size_t row = 0; row < buffer->rows; ++row)
    {
        int sequence_length = snprintf(sequence, sizeof(sequence), "\x1b[%zu;1H", row + 1);

        if (sequence_length < 0 || (size_t)sequence_length >= sizeof(sequence))
        {
            return false;
        }

        memcpy(renderer->output_buffer + position, sequence, (size_t)sequence_length);

        position += (size_t)sequence_length;

        for (size_t column = 0; column < buffer->columns; ++column)
        {
            const ScreenCell *cell = screen_buffer_get(buffer, row, column);

            if (cell == NULL)
            {
                return false;
            }

            char encoded[4];
            size_t encoded_length = renderer_encode_utf8(cell->code_point, encoded);

            if (encoded_length == 0)
            {
                return false;
            }

            memcpy(renderer->output_buffer + position, encoded, encoded_length);

            position += encoded_length;
        }
    }

    return renderer_write_all(renderer->output_buffer, frame_length);
}

static bool renderer_write_all(const char *data, size_t length)
{
    if (data == NULL)
    {
        return false;
    }

    size_t written = 0;

    while (written < length)
    {
        ssize_t result;

        /* Ask the OS to write the remaining bytes. */
        result = write(STDOUT_FILENO, data + written, length - written);

        if (result < 0)
        {
            /* EINTR means a signal interrupted us; retrying is safe. */
            if (errno == EINTR)
            {
                continue;
            }

            return false;
        }

        /* A return value of 0 means nothing could be written; give up. */
        if (result == 0)
        {
            return false;
        }

        written += (size_t)result;
    }

    return true;
}

static bool renderer_write_cursor_position(size_t row, size_t column)
{
    char sequence[64];
    int sequence_length = snprintf(sequence, sizeof(sequence), "\x1b[%zu;%zuH", row + 1, column + 1);

    if (sequence_length < 0 || (size_t)sequence_length >= sizeof(sequence))
    {
        return false;
    }

    return renderer_write_all(sequence, (size_t)sequence_length);
}

/*
 * Return the number of bytes required to encode a Unicode scalar value
 * as UTF-8. Invalid scalar values are rejected.
 */
static size_t renderer_utf8_length(uint32_t code_point)
{
    if (code_point <= UINT32_C(0x7F))
    {
        return 1;
    }

    if (code_point <= UINT32_C(0x7FF))
    {
        return 2;
    }

    if (code_point >= UINT32_C(0xD800) && code_point <= UINT32_C(0xDFFF))
    {
        return 0;
    }

    if (code_point <= UINT32_C(0xFFFF))
    {
        return 3;
    }

    if (code_point <= UINT32_C(0x10FFFF))
    {
        return 4;
    }

    return 0;
}

/*
 * Encode one Unicode scalar value into a four-byte output array.
 * The returned length is zero if the value is invalid.
 */
static size_t renderer_encode_utf8(uint32_t code_point, char output[4])
{
    size_t length = renderer_utf8_length(code_point);

    if (length == 0 || output == NULL)
    {
        return 0;
    }

    if (length == 1)
    {
        output[0] = (char)code_point;
    }
    else if (length == 2)
    {
        output[0] = (char)(UINT32_C(0xC0) | (code_point >> 6));

        output[1] = (char)(UINT32_C(0x80) | (code_point & UINT32_C(0x3F)));
    }
    else if (length == 3)
    {
        output[0] = (char)(UINT32_C(0xE0) | (code_point >> 12));

        output[1] = (char)(UINT32_C(0x80) | ((code_point >> 6) & UINT32_C(0x3F)));

        output[2] = (char)(UINT32_C(0x80) | (code_point & UINT32_C(0x3F)));
    }
    else
    {
        output[0] = (char)(UINT32_C(0xF0) | (code_point >> 18));

        output[1] = (char)(UINT32_C(0x80) | ((code_point >> 12) & UINT32_C(0x3F)));

        output[2] = (char)(UINT32_C(0x80) | ((code_point >> 6) & UINT32_C(0x3F)));

        output[3] = (char)(UINT32_C(0x80) | (code_point & UINT32_C(0x3F)));
    }

    return length;
}
```

```c
/* src/main.c */

#include "command/command_engine.h"
#include "cursor/cursor.h"
#include "document/document.h"
#include "input/input_engine.h"
#include "input/input_parser.h"
#include "renderer/renderer.h"
#include "state/editor_state.h"
#include "terminal/terminal.h"

#include <errno.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
#include <sys/ioctl.h>
#include <sys/select.h>
#include <unistd.h>

#define ESCAPE_TIMEOUT_MICROSECONDS 50000

static bool get_terminal_size(size_t *rows, size_t *columns)
{
    struct winsize window_size;

    if (rows == NULL || columns == NULL ||
        ioctl(STDOUT_FILENO, TIOCGWINSZ, &window_size) == -1 ||
        window_size.ws_row == 0 || window_size.ws_col == 0)
    {
        return false;
    }

    *rows = window_size.ws_row;
    *columns = window_size.ws_col;
    return true;
}

/* Read one byte while allowing an incomplete Escape sequence to time out. */
static int read_escape_byte(unsigned char *byte)
{
    fd_set read_set;
    struct timeval timeout;
    int result;

    FD_ZERO(&read_set);
    FD_SET(STDIN_FILENO, &read_set);
    timeout.tv_sec = 0;
    timeout.tv_usec = ESCAPE_TIMEOUT_MICROSECONDS;

    do
    {
        result = select(STDIN_FILENO + 1, &read_set, NULL, NULL, &timeout);
    } while (result == -1 && errno == EINTR);

    if (result <= 0)
    {
        return result;
    }

    for (;;)
    {
        const ssize_t bytes_read = read(STDIN_FILENO, byte, sizeof(*byte));

        if (bytes_read == 1)
        {
            return 1;
        }
        if (bytes_read == 0)
        {
            return 0;
        }
        if (errno != EINTR)
        {
            return -1;
        }
    }
}

static bool redraw_after_action(Renderer *renderer, EditorState *state)
{
    if (renderer == NULL || state == NULL)
    {
        return false;
    }

    if (!viewport_ensure_cursor_visible(
            &state->viewport,
            state->cursor.row,
            state->cursor.column,
            renderer->document_rows,
            renderer->screen_buffer.columns))
    {
        return false;
    }

    return renderer_render(renderer, state);
}

static bool application_event_loop(EditorState *state, Renderer *renderer, const char *save_path)
{
    InputParser parser;
    input_parser_init(&parser);

    while (true)
    {
        unsigned char byte;
        int read_result;
        InputEvent event;
        bool should_quit;
        bool event_ready = false;

        if (input_parser_is_waiting(&parser))
        {
            read_result = read_escape_byte(&byte);
            if (read_result == 0)
            {
                if (!input_parser_timeout(&parser, &event))
                {
                    return false;
                }
                if (event.type == INPUT_EVENT_NONE)
                {
                    continue;
                }
                event_ready = true;
                read_result = 1;
            }
        }
        else
        {
            const ssize_t bytes_read = read(STDIN_FILENO, &byte, sizeof(byte));
            if (bytes_read < 0)
            {
                if (errno == EINTR)
                {
                    continue;
                }
                return false;
            }
            if (bytes_read == 0)
            {
                return true;
            }
            read_result = 1;
        }

        if (read_result < 0)
        {
            return false;
        }
        if (read_result == 0)
        {
            return true;
        }
        if (!event_ready && !input_parser_feed(&parser, byte, &event))
        {
            return false;
        }
        EditorCommand command;
        bool redraw;
        if (!input_engine_translate(&event, &command))
        {
            return false;
        }
        if (!command_engine_execute(state, &command, save_path, &redraw, &should_quit))
        {
            return false;
        }
        if (redraw && !redraw_after_action(renderer, state))
        {
            return false;
        }
        if (should_quit)
        {
            return true;
        }
    }
}

int main(int argc, char *argv[])
{
    if (argc > 2)
    {
        fprintf(stderr, "Usage: %s [filename]\n", argv[0]);
        return 1;
    }

    const char *startup_path = argc == 2 ? argv[1] : NULL;
    bool startup_load_failed = false;
    TerminalState terminal;
    EditorState state;
    Renderer renderer;
    size_t rows;
    size_t columns;
    bool state_initialized = false;
    bool renderer_initialized = false;
    int exit_status = 1;

    if (!terminal_init(&terminal))
    {
        return 1;
    }
    if (!terminal_enable_raw_mode(&terminal))
    {
        terminal_restore(&terminal);
        return 1;
    }
    if (!terminal_enter_alternate_screen(&terminal))
    {
        terminal_restore(&terminal);
        return 1;
    }
    if (!get_terminal_size(&rows, &columns))
    {
        goto cleanup;
    }
    if (!editor_state_init(&state))
    {
        goto cleanup;
    }
    state_initialized = true;

    if (startup_path != NULL && !editor_state_load_file(&state, startup_path))
    {
        startup_load_failed = true;
        goto cleanup;
    }
    if (columns < 2 || !renderer_init(&renderer, rows, columns - 1))
    {
        goto cleanup;
    }
    renderer_initialized = true;

    if (!redraw_after_action(&renderer, &state) ||
        !application_event_loop(&state, &renderer, startup_path))
    {
        goto cleanup;
    }

    exit_status = 0;

cleanup:
    if (renderer_initialized)
    {
        renderer_destroy(&renderer);
    }
    if (state_initialized)
    {
        editor_state_destroy(&state);
    }
    terminal_restore(&terminal);
    terminal_leave_alternate_screen(&terminal);

    if (startup_load_failed)
    {
        fprintf(stderr, "tedit: unable to open or decode '%s'.\n", startup_path);
    }

    return exit_status;
}
```

```c
#include "input/input_parser.h"

#include <assert.h>
#include <stddef.h>

static InputEvent feed(InputParser *parser, unsigned char byte)
{
    InputEvent event;
    assert(input_parser_feed(parser, byte, &event));
    return event;
}

static void test_ascii_and_control_keys(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    event = feed(&parser, 'x');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'x');
    event = feed(&parser, 0x11);
    assert(event.type == INPUT_EVENT_CTRL_Q);
    event = feed(&parser, 0x13);
    assert(event.type == INPUT_EVENT_CTRL_S);
    event = feed(&parser, 0x7F);
    assert(event.type == INPUT_EVENT_BACKSPACE);
    event = feed(&parser, 0x08);
    assert(event.type == INPUT_EVENT_BACKSPACE);
    event = feed(&parser, '\r');
    assert(event.type == INPUT_EVENT_ENTER);
    event = feed(&parser, '\n');
    assert(event.type == INPUT_EVENT_ENTER);
}

static void test_arrow_sequences(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    event = feed(&parser, 0x1B);
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_is_waiting(&parser));
    event = feed(&parser, '[');
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_is_waiting(&parser));
    event = feed(&parser, 'A');
    assert(event.type == INPUT_EVENT_ARROW_UP);
    assert(!input_parser_is_waiting(&parser));

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'B');
    assert(event.type == INPUT_EVENT_ARROW_DOWN);
```

```c
static void feed_and_execute(InputParser *parser, EditorState *state, unsigned char byte,
                             bool *redraw, bool *should_quit)
{
    InputEvent event;
    EditorCommand command;

    assert(input_parser_feed(parser, byte, &event));
    assert(input_engine_translate(&event, &command));
    assert(command_engine_execute(state, &command, NULL, redraw, should_quit));
}

static void test_end_to_end_input_pipeline(void)
{
    InputParser parser;
    EditorState state;
    bool redraw;
    bool should_quit;
    const DocumentLine *line;

    assert(editor_state_init(&state));
    input_parser_init(&parser);

    feed_and_execute(&parser, &state, 'x', &redraw, &should_quit);
    assert(redraw && !should_quit);
    assert(state.is_dirty);
    line = document_get_line(&state.document, 0);
    assert(line != NULL && line->length == 1 && line->data[0] == 'x');
    assert(state.cursor.column == 1);

    /* An unknown CSI sequence produces no command and does not change state. */
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    assert(!redraw && !should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    assert(!redraw && !should_quit);
    feed_and_execute(&parser, &state, 'Z', &redraw, &should_quit);
    assert(!redraw && !should_quit);
    assert(state.cursor.column == 1);

    /* A complete arrow sequence is parsed before it becomes a command. */
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, 'D', &redraw, &should_quit);
    assert(redraw && !should_quit);
    assert(state.cursor.column == 0);

    /* Save without an associated path starts filename entry. */
    feed_and_execute(&parser, &state, 0x13, &redraw, &should_quit);
    assert(redraw && !should_quit);
    assert(state.is_dirty);
    assert(state.filename_entry_active);
    assert(strcmp(state.message, "Enter: save  Esc: cancel") == 0);

    /* Escape is recognized after the parser's standalone-key timeout. */
    InputEvent event;
    EditorCommand command;
    assert(input_parser_feed(&parser, 0x1B, &event));
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_ESCAPE);
    assert(input_engine_translate(&event, &command));
    assert(command.type == EDITOR_COMMAND_CANCEL);
    assert(command_engine_execute(&state, &command, NULL, &redraw, &should_quit));
    assert(redraw && !should_quit && !state.filename_entry_active);
    assert(state.is_dirty);

    feed_and_execute(&parser, &state, 0x11, &redraw, &should_quit);
    assert(!redraw && should_quit);

    editor_state_destroy(&state);
}

int main(void)
{
    test_end_to_end_input_pipeline();
    return 0;
}
```

```c
static void write_all(int descriptor, const unsigned char *data, size_t length)
{
    size_t offset = 0;

    while (offset < length)
    {
        ssize_t written = write(descriptor, data + offset, length - offset);
        assert(written > 0);
        offset += (size_t)written;
    }
}

    assert(!ferror(file));

    return length;
```

## Verification

All eight CTest targets passed in both Debug and warning-clean builds: `smoke`, `editing`, `file`, `file_save`, `input_parser`, `input_engine`, `command_engine`, and `input_pipeline`. Pseudo-terminal integration checks verified that typing a document, entering a destination after Ctrl-S, and pressing Enter writes the file and exits cleanly with Ctrl-Q. A separate pseudo-terminal check verified Escape cancellation followed by Ctrl-Q.

## Current limitation

UTF-8 filename bytes are preserved for filesystem operations, but the current Status Bar and filename prompt render non-ASCII bytes as `?`. Ctrl-S on a document that already has an associated path retains its existing save-to-current-file behavior; a separate explicit Save As command for already-associated documents is not included in this milestone.
