## Implementation Goal

Ensure terminal UTF-8 input is converted into complete Unicode scalar values before it reaches the editing pipeline, and verify that navigation and deletion operate on logical code points rather than UTF-8 bytes.

static InputEvent feed(InputParser *parser, unsigned char byte)
{
    InputEvent event;
    assert(input_parser_feed(parser, byte, &event));
    return event;
}

static void test_arrow_sequences(void)
{

    assert(input_parser_is_waiting(&parser));

    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_is_waiting(&parser));
    event = feed(&parser, 'A');
    assert(event.type == INPUT_EVENT_ARROW_UP);

    event = feed(&parser, 0x1B);

    event = feed(&parser, 'C');

    event = feed(&parser, 0x1B);

    event = feed(&parser, 'D');

static void test_escape_timeout_and_recovery(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    event = feed(&parser, 0x1B);
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_ESCAPE);
    assert(!input_parser_is_waiting(&parser));

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 'z');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'z');

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'X');
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 'q');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'q');
}

static void test_utf8_scalar_boundaries_and_recovery(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);

    /* Two-, three-, and four-byte sequences each emit one scalar event. */
    event = feed(&parser, 0xC3);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xA9);
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 0x00E9);

    event = feed(&parser, 0xE2);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x82);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xAC);
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 0x20AC);

    event = feed(&parser, 0xF0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x9F);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x98);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 0x1F600);

    /* Invalid leading bytes, overlong forms, surrogates, and out-of-range
     * values must never become character events. */
    event = feed(&parser, 0xC0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xAF);
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 0xE0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 0xED);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xA0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 0xF4);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x90);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);

    /* A non-continuation byte aborts the incomplete sequence and is parsed
     * again as a new input byte instead of being swallowed. */
    event = feed(&parser, 0xE2);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 'r');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'r');
    assert(parser.utf8_remaining == 0);

    /* Timeout also clears an incomplete UTF-8 sequence. */
    event = feed(&parser, 0xF0);
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_NONE && parser.utf8_remaining == 0);
    event = feed(&parser, 's');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 's');
}

int main(void)
{
    test_ascii_and_control_keys();
    test_arrow_sequences();
    test_home_end_and_page_sequences();
    test_escape_timeout_and_recovery();
    test_utf8_character();
    test_invalid_utf8_recovers();
    test_utf8_scalar_boundaries_and_recovery();
    return 0;
}

static void execute(EditorState *state, EditorCommandType type, uint32_t character,

static void test_editing_commands(void)
{

    bool should_quit;

    assert(redraw && state.document.line_count == 2 && state.cursor.row == 1);
    editor_state_destroy(&state);
}

static void test_page_navigation_commands(void)
{
    EditorState state;
    bool redraw;
    bool should_quit;

    assert(editor_state_init(&state));

    /* Build eight logical lines, each with four code points. */
    for (size_t row = 0; row < 8; row++)
    {
        for (size_t column = 0; column < 4; column++)
        {
            execute(&state, EDITOR_COMMAND_INSERT_CHARACTER, (uint32_t)('a' + column),
                    &redraw, &should_quit);
        }
        if (row + 1 < 8)
        {
            execute(&state, EDITOR_COMMAND_INSERT_NEWLINE, 0, &redraw, &should_quit);
        }
    }

    assert(state.cursor.row == 7 && state.cursor.column == 4);

    /* Make one destination line shorter to verify preferred-column recovery. */
    assert(document_delete_character(&state.document, 4, 1));
    assert(document_delete_character(&state.document, 4, 1));
    assert(document_delete_character(&state.document, 4, 1));
    state.is_dirty = false;

    execute(&state, EDITOR_COMMAND_PAGE_UP, 0, &redraw, &should_quit);
    assert(redraw && !should_quit);
    assert(state.cursor.row == 4 && state.cursor.column == 1);
    assert(state.cursor.preferred_column == 4);
    assert(!state.is_dirty);

    execute(&state, EDITOR_COMMAND_PAGE_UP, 0, &redraw, &should_quit);
    assert(redraw && state.cursor.row == 1 && state.cursor.column == 4);
    execute(&state, EDITOR_COMMAND_PAGE_UP, 0, &redraw, &should_quit);
    assert(redraw && state.cursor.row == 0 && state.cursor.column == 4);
    execute(&state, EDITOR_COMMAND_PAGE_UP, 0, &redraw, &should_quit);
    assert(!redraw && state.cursor.row == 0);

    execute(&state, EDITOR_COMMAND_PAGE_DOWN, 0, &redraw, &should_quit);
    assert(redraw && state.cursor.row == 3 && state.cursor.column == 4);
    execute(&state, EDITOR_COMMAND_PAGE_DOWN, 0, &redraw, &should_quit);
    assert(redraw && state.cursor.row == 6 && state.cursor.column == 4);
    execute(&state, EDITOR_COMMAND_PAGE_DOWN, 0, &redraw, &should_quit);
    assert(redraw && state.cursor.row == 7 && state.cursor.column == 4);
    execute(&state, EDITOR_COMMAND_PAGE_DOWN, 0, &redraw, &should_quit);
    assert(!redraw && state.cursor.row == 7);
    assert(!state.is_dirty);

    editor_state_destroy(&state);
}

static void feed_and_execute(InputParser *parser, EditorState *state, unsigned char byte,
                             bool *redraw, bool *should_quit)
{
    InputEvent event;
    EditorCommand command;

    assert(input_parser_feed(parser, byte, &event));
    assert(input_engine_translate(&event, &command));
    command.page_rows = 3;
    assert(command_engine_execute(state, &command, NULL, redraw, should_quit));
}

static void test_end_to_end_input_pipeline(void)
{
    InputParser parser;
    EditorState state;
    bool redraw;
    bool should_quit;
    const DocumentLine *line;

    assert(editor_state_init(&state));
    input_parser_init(&parser);

    feed_and_execute(&parser, &state, 'x', &redraw, &should_quit);
    assert(redraw && !should_quit);
    assert(state.is_dirty);
    line = document_get_line(&state.document, 0);
    assert(line != NULL && line->length == 1 && line->data[0] == 'x');
    assert(state.cursor.column == 1);

    /* An unknown CSI sequence produces no command and does not change state. */
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    assert(!redraw && !should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    assert(!redraw && !should_quit);
    feed_and_execute(&parser, &state, 'Z', &redraw, &should_quit);
    assert(!redraw && !should_quit);
    assert(state.cursor.column == 1);

    /* A complete arrow sequence is parsed before it becomes a command. */
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, 'D', &redraw, &should_quit);
    assert(redraw && !should_quit);
    assert(state.cursor.column == 0);

    /* Home and End terminal sequences reach document navigation commands. */
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, 'C', &redraw, &should_quit);
    assert(redraw && state.cursor.column == 1);
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, 'H', &redraw, &should_quit);
    assert(redraw && state.cursor.column == 0);
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, 'F', &redraw, &should_quit);
    assert(redraw && state.cursor.column == 1);

    /* Delete removes the code point at the Cursor through the full pipeline. */
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, 'H', &redraw, &should_quit);
    feed_and_execute(&parser, &state, 0x1B, &redraw, &should_quit);
    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, '3', &redraw, &should_quit);
    feed_and_execute(&parser, &state, '~', &redraw, &should_quit);
    line = document_get_line(&state.document, 0);
    assert(redraw && line != NULL && line->length == 0);
    assert(state.cursor.row == 0 && state.cursor.column == 0 && state.is_dirty);

    /* Save without an associated path starts filename entry. */
    feed_and_execute(&parser, &state, 0x13, &redraw, &should_quit);
    assert(redraw && !should_quit);
    assert(state.is_dirty);
    assert(state.filename_entry_active);
    assert(strcmp(state.message, "Enter: save  Esc: cancel") == 0);

    /* Escape is recognized after the parser's standalone-key timeout. */
    InputEvent event;
    EditorCommand command;
    assert(input_parser_feed(&parser, 0x1B, &event));
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_ESCAPE);
    assert(input_engine_translate(&event, &command));
    assert(command.type == EDITOR_COMMAND_CANCEL);
    assert(command_engine_execute(&state, &command, NULL, &redraw, &should_quit));
    assert(redraw && !should_quit && !state.filename_entry_active);
    assert(state.is_dirty);

    feed_and_execute(&parser, &state, 0x11, &redraw, &should_quit);
    assert(redraw && !should_quit && state.quit_confirmation_active);
    assert(strstr(state.message, "Save changes before quitting?") != NULL);
    feed_and_execute(&parser, &state, 'n', &redraw, &should_quit);
    assert(!redraw && should_quit && !state.quit_confirmation_active);

    editor_state_destroy(&state);
}

static void test_page_navigation_pipeline(void)
{

    feed_and_execute(&parser, &state, '[', &redraw, &should_quit);
    feed_and_execute(&parser, &state, '6', &redraw, &should_quit);

    editor_state_destroy(&state);
}

    bool redraw;
    bool should_quit;

    input_parser_init(&parser);

    line = document_get_line(&state.document, 0);
    assert(redraw && line != NULL && line->length == 0);
    assert(state.cursor.column == 0);

}

- Added parser tests for two-byte, three-byte, and four-byte UTF-8 sequences.
- Added rejection tests for invalid leading bytes, overlong encodings, UTF-16 surrogate values, and code points above U+10FFFF.
- Verified that a non-continuation byte interrupts and clears a partial sequence, then is reprocessed as a new input byte.
- Verified that parser timeout clears incomplete UTF-8 state.
- Added Command Engine coverage proving Backspace deletes one stored Unicode code point at a time.
- Added full input-pipeline coverage proving a four-byte UTF-8 character inserts one logical Document code point and can be removed with one Backspace operation.
- Kept the Document and Cursor model unchanged: line lengths and cursor columns count code points, not UTF-8 bytes.
