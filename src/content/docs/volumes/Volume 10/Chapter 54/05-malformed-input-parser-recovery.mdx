### `src/input/input_parser.c`

```c
/* src/input/input_parser.c */

#include "input/input_parser.h"

#include <stddef.h>

#define INPUT_CONTROL_Q 0x11
#define INPUT_CONTROL_S 0x13
#define INPUT_BACKSPACE 0x7F
#define INPUT_BACKSPACE_ALT 0x08
#define INPUT_ESCAPE 0x1B
#define INPUT_ENTER 0x0D
#define INPUT_TAB 0x09
#define INPUT_LINE_FEED 0x0A
#define UTF8_CONTINUATION_MASK 0xC0
#define UTF8_CONTINUATION_PREFIX 0x80
#define UTF8_TWO_BYTE_PREFIX 0xC0
#define UTF8_THREE_BYTE_PREFIX 0xE0
#define UTF8_FOUR_BYTE_PREFIX 0xF0

void input_parser_init(InputParser *parser)
{
    if (parser == NULL)
    {
        return;
    }

    parser->state = INPUT_PARSER_NORMAL;
    parser->utf8_code_point = 0;
    parser->utf8_minimum = 0;
    parser->utf8_remaining = 0;
    parser->csi_parameter = 0;
}

static void input_event_clear(InputEvent *event)
{
    event->type = INPUT_EVENT_NONE;
    event->character = 0;
}

static bool input_parser_feed_utf8(InputParser *parser, unsigned char byte, InputEvent *event)
{
    if (parser->utf8_remaining > 0)
    {
        if ((byte & UTF8_CONTINUATION_MASK) != UTF8_CONTINUATION_PREFIX)
        {
            parser->utf8_remaining = 0;
            parser->utf8_code_point = 0;
            parser->utf8_minimum = 0;
            return input_parser_feed(parser, byte, event);
        }

        parser->utf8_code_point = (parser->utf8_code_point << 6) | (uint32_t)(byte & 0x3F);
        parser->utf8_remaining--;

        if (parser->utf8_remaining == 0)
        {
            const uint32_t code_point = parser->utf8_code_point;
            const bool valid = code_point >= parser->utf8_minimum &&
                               code_point <= 0x10FFFF &&
                               !(code_point >= 0xD800 && code_point <= 0xDFFF);
            parser->utf8_code_point = 0;
            parser->utf8_minimum = 0;

            if (valid)
            {
                event->type = INPUT_EVENT_CHARACTER;
                event->character = code_point;
            }
        }

        return true;
    }

    if (byte >= 0xC2 && byte <= 0xDF)
    {
        parser->utf8_code_point = (uint32_t)(byte & 0x1F);
        parser->utf8_minimum = 0x80;
        parser->utf8_remaining = 1;
    }
    else if (byte >= 0xE0 && byte <= 0xEF)
    {
        parser->utf8_code_point = (uint32_t)(byte & 0x0F);
        parser->utf8_minimum = 0x800;
        parser->utf8_remaining = 2;
    }
    else if (byte >= 0xF0 && byte <= 0xF4)
    {
        parser->utf8_code_point = (uint32_t)(byte & 0x07);
        parser->utf8_minimum = 0x10000;
        parser->utf8_remaining = 3;
    }

    return true;
}

bool input_parser_feed(InputParser *parser, unsigned char byte, InputEvent *event)
{
    if (parser == NULL || event == NULL)
    {
        return false;
    }

    input_event_clear(event);

    if (parser->utf8_remaining > 0)
    {
        return input_parser_feed_utf8(parser, byte, event);
    }

    if (parser->state == INPUT_PARSER_NORMAL)
    {
        if (byte == INPUT_ESCAPE)
        {
            parser->state = INPUT_PARSER_ESCAPE;
        }
        else if (byte == INPUT_CONTROL_Q)
        {
            event->type = INPUT_EVENT_CTRL_Q;
        }
        else if (byte == INPUT_CONTROL_S)
        {
            event->type = INPUT_EVENT_CTRL_S;
        }
        else if (byte == INPUT_BACKSPACE || byte == INPUT_BACKSPACE_ALT)
        {
            event->type = INPUT_EVENT_BACKSPACE;
        }
        else if (byte == INPUT_ENTER || byte == INPUT_LINE_FEED)
        {
            event->type = INPUT_EVENT_ENTER;
        }
        else if (byte == INPUT_TAB)
        {
            event->type = INPUT_EVENT_TAB;
        }
        else if (byte >= 0x20 && byte <= 0x7E)
        {
            event->type = INPUT_EVENT_CHARACTER;
            event->character = byte;
        }
        else if (byte >= 0x80)
        {
            return input_parser_feed_utf8(parser, byte, event);
        }

        return true;
    }

    if (parser->state == INPUT_PARSER_ESCAPE)
    {
        if (byte == '[')
        {
            parser->state = INPUT_PARSER_CSI;
            parser->csi_parameter = 0;
        }
        else if (byte == INPUT_ESCAPE)
        {
            parser->state = INPUT_PARSER_ESCAPE;
        }
        else
        {
            parser->state = INPUT_PARSER_NORMAL;
        }

        return true;
    }

    if (parser->state == INPUT_PARSER_CSI_PARAMETER)
    {
        if (byte >= '0' && byte <= '9')
        {
            if (parser->csi_parameter <= 999)
            {
                parser->csi_parameter = parser->csi_parameter * 10U + (unsigned int)(byte - '0');
            }
            return true;
        }

        if (byte == '~')
        {
            if (parser->csi_parameter == 1 || parser->csi_parameter == 7)
            {
                event->type = INPUT_EVENT_HOME;
            }
            else if (parser->csi_parameter == 4 || parser->csi_parameter == 8)
            {
                event->type = INPUT_EVENT_END;
            }
            else if (parser->csi_parameter == 3)
            {
                event->type = INPUT_EVENT_DELETE;
            }
            else if (parser->csi_parameter == 5)
            {
                event->type = INPUT_EVENT_PAGE_UP;
            }
            else if (parser->csi_parameter == 6)
            {
                event->type = INPUT_EVENT_PAGE_DOWN;
            }
        }
        parser->state = INPUT_PARSER_NORMAL;
        parser->csi_parameter = 0;

        /* A control byte can be a real key even if it interrupts a malformed
         * CSI parameter. Reprocess it after abandoning the incomplete sequence
         * so Ctrl-S, Ctrl-Q, Escape, and similar keys are not swallowed. */
        if (byte < 0x20 || byte == 0x7F)
        {
            return input_parser_feed(parser, byte, event);
        }

        return true;
    }

    if (byte >= '0' && byte <= '9')
    {
        parser->state = INPUT_PARSER_CSI_PARAMETER;
        parser->csi_parameter = (unsigned int)(byte - '0');
        return true;
    }

    parser->state = INPUT_PARSER_NORMAL;
    parser->csi_parameter = 0;

    /* Do not consume a standalone control key merely because it interrupted
     * an escape sequence. Start parsing that byte again from NORMAL state. */
    if (byte < 0x20 || byte == 0x7F)
    {
        return input_parser_feed(parser, byte, event);
    }

    if (byte == 'D')
    {
        event->type = INPUT_EVENT_ARROW_LEFT;
    }
    else if (byte == 'H')
    {
        event->type = INPUT_EVENT_HOME;
    }
    else if (byte == 'F')
    {
        event->type = INPUT_EVENT_END;
    }
    else if (byte == 'C')
    {
        event->type = INPUT_EVENT_ARROW_RIGHT;
    }
    else if (byte == 'A')
    {
        event->type = INPUT_EVENT_ARROW_UP;
    }
    else if (byte == 'B')
    {
        event->type = INPUT_EVENT_ARROW_DOWN;
    }

    return true;
}

bool input_parser_timeout(InputParser *parser, InputEvent *event)
{
    if (parser == NULL || event == NULL)
    {
        return false;
    }

    const InputParserState previous_state = parser->state;

    parser->state = INPUT_PARSER_NORMAL;
    parser->utf8_code_point = 0;
    parser->utf8_minimum = 0;
    parser->utf8_remaining = 0;
    parser->csi_parameter = 0;
    input_event_clear(event);
    if (previous_state == INPUT_PARSER_ESCAPE)
    {
        event->type = INPUT_EVENT_ESCAPE;
    }
    return true;
}

bool input_parser_is_waiting(const InputParser *parser)
{
    return parser != NULL && parser->state != INPUT_PARSER_NORMAL;
}
```

### `tests/test_input_parser.c`

```c
/* tests/test_input_parser.c */

#include "input/input_parser.h"

#include <assert.h>
#include <stddef.h>

static InputEvent feed(InputParser *parser, unsigned char byte)
{
    InputEvent event;
    assert(input_parser_feed(parser, byte, &event));
    return event;
}

static void test_ascii_and_control_keys(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    event = feed(&parser, 'x');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'x');
    event = feed(&parser, 0x11);
    assert(event.type == INPUT_EVENT_CTRL_Q);
    event = feed(&parser, 0x13);
    assert(event.type == INPUT_EVENT_CTRL_S);
    event = feed(&parser, 0x7F);
    assert(event.type == INPUT_EVENT_BACKSPACE);
    event = feed(&parser, 0x08);
    assert(event.type == INPUT_EVENT_BACKSPACE);
    event = feed(&parser, '\r');
    assert(event.type == INPUT_EVENT_ENTER);
    event = feed(&parser, '\n');
    assert(event.type == INPUT_EVENT_ENTER);
    event = feed(&parser, '\t');
    assert(event.type == INPUT_EVENT_TAB);
}

static void test_arrow_sequences(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    event = feed(&parser, 0x1B);
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_is_waiting(&parser));
    event = feed(&parser, '[');
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_is_waiting(&parser));
    event = feed(&parser, 'A');
    assert(event.type == INPUT_EVENT_ARROW_UP);
    assert(!input_parser_is_waiting(&parser));

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'B');
    assert(event.type == INPUT_EVENT_ARROW_DOWN);
    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'C');
    assert(event.type == INPUT_EVENT_ARROW_RIGHT);
    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'D');
    assert(event.type == INPUT_EVENT_ARROW_LEFT);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '3');
    assert(event.type == INPUT_EVENT_NONE && input_parser_is_waiting(&parser));
    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_DELETE && !input_parser_is_waiting(&parser));
}


static void test_home_end_and_page_sequences(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'H');
    assert(event.type == INPUT_EVENT_HOME);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'F');
    assert(event.type == INPUT_EVENT_END);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '1');
    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_HOME);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '4');
    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_END);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '5');
    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_PAGE_UP);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '6');
    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_PAGE_DOWN);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '7');
    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_HOME);

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '8');
    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_END);
}

static void test_escape_timeout_and_recovery(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    event = feed(&parser, 0x1B);
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_ESCAPE);
    assert(!input_parser_is_waiting(&parser));

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 'z');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'z');

    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 'X');
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 'q');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'q');
}

    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    event = feed(&parser, 0xE2);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x82);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xAC);
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 0x20AC);
}

static void test_invalid_utf8_recovers(void)

    InputParser parser;

    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 'a');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'a');
}

static void test_utf8_scalar_boundaries_and_recovery(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);

    /* Two-, three-, and four-byte sequences each emit one scalar event. */
    event = feed(&parser, 0xC3);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xA9);
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 0x00E9);

    event = feed(&parser, 0xE2);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x82);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xAC);
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 0x20AC);

    event = feed(&parser, 0xF0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x9F);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x98);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 0x1F600);

    /* Invalid leading bytes, overlong forms, surrogates, and out-of-range
     * values must never become character events. */
    event = feed(&parser, 0xC0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xAF);
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 0xE0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 0xED);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0xA0);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);

    event = feed(&parser, 0xF4);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x90);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x80);
    assert(event.type == INPUT_EVENT_NONE);

    /* A non-continuation byte aborts the incomplete sequence and is parsed
     * again as a new input byte instead of being swallowed. */
    event = feed(&parser, 0xE2);
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 'r');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'r');
    assert(parser.utf8_remaining == 0);

    /* Timeout also clears an incomplete UTF-8 sequence. */
    event = feed(&parser, 0xF0);
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_NONE && parser.utf8_remaining == 0);
    event = feed(&parser, 's');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 's');
}


static void test_malformed_csi_control_recovery(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);

    /* Ctrl-S must not be swallowed by an incomplete numeric CSI sequence. */
    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, '3');
    assert(event.type == INPUT_EVENT_NONE);
    event = feed(&parser, 0x13);
    assert(event.type == INPUT_EVENT_CTRL_S);
    assert(!input_parser_is_waiting(&parser));

    event = feed(&parser, 'x');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'x');

    /* A fresh Escape byte restarts parsing after a malformed CSI prefix. */
    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');
    event = feed(&parser, 0x1B);
    assert(event.type == INPUT_EVENT_NONE);
    assert(input_parser_is_waiting(&parser));
    assert(input_parser_timeout(&parser, &event));
    assert(event.type == INPUT_EVENT_ESCAPE);
    assert(!input_parser_is_waiting(&parser));
}

static void test_oversized_csi_parameter_recovery(void)
{
    InputParser parser;
    InputEvent event;
    const char *digits = "999999999999999999999999999999";
    size_t i;

    input_parser_init(&parser);
    event = feed(&parser, 0x1B);
    event = feed(&parser, '[');

    for (i = 0; digits[i] != '\0'; ++i)
    {
        event = feed(&parser, (unsigned char)digits[i]);
        assert(event.type == INPUT_EVENT_NONE);
    }

    event = feed(&parser, '~');
    assert(event.type == INPUT_EVENT_NONE);
    assert(!input_parser_is_waiting(&parser));

    event = feed(&parser, 'r');
    assert(event.type == INPUT_EVENT_CHARACTER && event.character == 'r');
}

static void test_null_arguments_are_rejected(void)
{
    InputParser parser;
    InputEvent event;

    input_parser_init(&parser);
    assert(!input_parser_feed(NULL, 'x', &event));
    assert(!input_parser_feed(&parser, 'x', NULL));
    assert(!input_parser_timeout(NULL, &event));
    assert(!input_parser_timeout(&parser, NULL));
}

int main(void)
{
    test_ascii_and_control_keys();
    test_arrow_sequences();
    test_home_end_and_page_sequences();
    test_escape_timeout_and_recovery();
    test_utf8_character();
    test_invalid_utf8_recovers();
    test_utf8_scalar_boundaries_and_recovery();
    test_malformed_csi_control_recovery();
    test_oversized_csi_parameter_recovery();
    test_null_arguments_are_rejected();
    return 0;
}
```
