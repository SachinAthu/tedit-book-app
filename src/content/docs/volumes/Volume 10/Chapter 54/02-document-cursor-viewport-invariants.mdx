/*
 * Unit tests for the core editing behaviour.
 *
 * Each test function below exercises one small piece of behaviour (inserting

 * control) and uses assert() so any failure stops the program with a message.

/* Compare a line's code points against a NUL-terminated ASCII string. */
static bool line_equals(const DocumentLine *line, const char *text)
{
    size_t length = strlen(text);

    if (line == NULL || line->length != length)
    {
        return false;
    }

    for (size_t index = 0; index < length; ++index)
    {
        if (line->data[index] != (uint32_t)(unsigned char)text[index])
        {
            return false;
        }
    }

    return true;
}

/* Check the structural invariants owned by the Document module. */
static void assert_document_invariant(const Document *document)
{
    assert(document != NULL);
    assert(document->line_count <= document->line_capacity);
    assert((document->lines == NULL) == (document->line_capacity == 0));

    for (size_t row = 0; row < document->line_count; ++row)
    {
        const DocumentLine *line = &document->lines[row];

        assert(line->data != NULL);
        assert(line->capacity > 0);
        assert(line->length <= line->capacity);
    }
}

/* Check that the Cursor denotes a valid insertion position. */
static void assert_cursor_invariant(const Cursor *cursor, const Document *document)
{
    assert(cursor != NULL);
    assert_document_invariant(document);

    if (document->line_count == 0)
    {
        assert(cursor->row == 0);
        assert(cursor->column == 0);
        return;
    }

    assert(cursor->row < document->line_count);
    assert(cursor->column <= document->lines[cursor->row].length);
}

/* Check that viewport adjustment leaves the requested Cursor visible. */
static void assert_cursor_visible(
    const Viewport *viewport,
    size_t cursor_row,
    size_t cursor_visual_column,
    size_t screen_rows,
    size_t screen_columns)
{
    assert(viewport != NULL);
    assert(screen_rows > 0);
    assert(screen_columns > 0);
    assert(cursor_row >= viewport->viewport_row);
    assert(cursor_row - viewport->viewport_row < screen_rows);
    assert(cursor_visual_column >= viewport->viewport_column);
    assert(cursor_visual_column - viewport->viewport_column < screen_columns);
}

/* Invalid Document operations must fail without corrupting valid contents. */
static void test_document_invalid_operations_preserve_invariants(void)
{
    Document document;
    DocumentLine *line;

    assert(document_init(&document));
    assert(document_insert_character(&document, 0, 0, 'a'));
    assert(document_insert_character(&document, 0, 1, 'b'));
    assert(document_split_line(&document, 0, 2));
    assert(document_insert_character(&document, 1, 0, 'c'));
    assert_document_invariant(&document);

    assert(!document_insert_character(&document, 2, 0, 'x'));
    assert(!document_insert_character(&document, 0, 3, 'x'));
    assert(!document_insert_character(&document, 0, 1, UINT32_C(0xD800)));
    assert(!document_insert_character(&document, 0, 1, UINT32_C(0x110000)));
    assert(!document_delete_character(&document, 0, 2));
    assert(!document_delete_character(&document, 3, 0));
    assert(!document_split_line(&document, 0, 3));
    assert(!document_split_line(&document, 3, 0));
    assert(!document_merge_lines(&document, 1));
    assert(!document_merge_lines(&document, 99));

    assert_document_invariant(&document);
    line = document_get_line(&document, 0);
    assert(line_equals(line, "ab"));
    line = document_get_line(&document, 1);
    assert(line_equals(line, "c"));

    document_destroy(&document);
    assert(document.line_count == 0);
    assert(document.lines == NULL);
    assert(document.line_capacity == 0);
}

/* Failed Cursor placement leaves its previously valid position untouched. */
static void test_cursor_invalid_positions_preserve_invariants(void)
{
    Document document;

    assert(document_init(&document));

/* Cursor visibility remains an invariant after repeated viewport movement. */
static void test_viewport_visibility_invariant(void)
{
    Viewport viewport;

    viewport_init(&viewport);
    for (size_t row = 0; row < 20; ++row)
    {
        for (size_t column = 0; column < 30; ++column)
        {
            assert(viewport_ensure_cursor_visible(&viewport, row, column, 4, 7));
            assert_cursor_visible(&viewport, row, column, 4, 7);
        }
    }

    for (size_t row = 20; row > 0; --row)
    {
        for (size_t column = 30; column > 0; --column)
        {
            assert(viewport_ensure_cursor_visible(&viewport, row - 1, column - 1, 4, 7));
            assert_cursor_visible(&viewport, row - 1, column - 1, 4, 7);
        }
    }
}

    assert(document_insert_character(&document, 1, 0, '1'));

    cursor_init(&cursor);
    assert(cursor_set_position(&cursor, &document, 0, 3));
    assert(cursor_move_down(&cursor, &document, DEFAULT_TAB_WIDTH));
    assert(cursor.row == 1 && cursor.column == 4);

    assert(cursor.row == 1 && cursor.column == 4);
    assert(cursor.preferred_visual_column == 5);
    assert(cursor_move_up(&cursor, &document, DEFAULT_TAB_WIDTH));
    assert(cursor.row == 0 && cursor.column == 3);

    document_destroy(&document);
}

/* Repeated scrolling and screen edges keep the cursor inside the viewport. */
static void test_viewport_repeated_scrolling_and_screen_boundaries(void)
{
    Viewport viewport;
    const size_t screen_rows = 3;
    const size_t screen_columns = 4;
    size_t row;
    size_t column;

    viewport_init(&viewport);

    /* Every cursor position remains inside the viewport after adjustment. */
    for (row = 0; row < 10; ++row)
    {
        for (column = 0; column < 12; ++column)
        {
            assert(viewport_ensure_cursor_visible(
                &viewport, row, column, screen_rows, screen_columns));
            assert(row >= viewport.viewport_row);
            assert(row - viewport.viewport_row < screen_rows);
            assert(column >= viewport.viewport_column);
            assert(column - viewport.viewport_column < screen_columns);
        }
    }

    /* Moving back toward the document origin scrolls both axes back. */
    assert(viewport_ensure_cursor_visible(&viewport, 0, 0,
                                          screen_rows, screen_columns));
    assert(viewport.viewport_row == 0);
    assert(viewport.viewport_column == 0);

    /* The last visible cell does not force an unnecessary scroll. */
    assert(viewport_ensure_cursor_visible(&viewport, 2, 3,
                                          screen_rows, screen_columns));
    assert(viewport.viewport_row == 0);
    assert(viewport.viewport_column == 0);

    /* One position beyond the bottom-right visible cell scrolls by one. */
    assert(viewport_ensure_cursor_visible(&viewport, 3, 4,
                                          screen_rows, screen_columns));
    assert(viewport.viewport_row == 1);
    assert(viewport.viewport_column == 1);
}

/* The renderer uses the viewport's row/column offsets when drawing. */
static void test_renderer_uses_viewport_offsets(void)
{
    EditorState state;
    Renderer renderer;
    int output_pipe[2];
    int saved_stdout;
    char output[2048];
    ssize_t output_length;
    const ScreenCell *cell;

    assert(editor_state_init(&state));
    assert(document_insert_character(&state.document, 0, 0, 'a'));
    assert(document_insert_character(&state.document, 0, 1, 'b'));
    assert(document_insert_character(&state.document, 0, 2, 'c'));
    assert(document_split_line(&state.document, 0, 3));
    assert(document_insert_character(&state.document, 1, 0, 'd'));
    assert(document_insert_character(&state.document, 1, 1, 'e'));
    assert(document_insert_character(&state.document, 1, 2, 'f'));
    assert(document_insert_character(&state.document, 1, 3, 'g'));
    assert(document_insert_character(&state.document, 1, 4, 'h'));
    assert(cursor_set_position(&state.cursor, &state.document, 1, 3));
    assert(viewport_ensure_cursor_visible(&state.viewport, 1, 3, 2, 3));
    assert(state.viewport.viewport_row == 0);
    assert(state.viewport.viewport_column == 1);

    /* Force a vertical offset too, as if the document had been scrolled. */
    state.viewport.viewport_row = 1;
    state.viewport.viewport_column = 2;

    assert(renderer_init(&renderer, 4, 3));
    assert(pipe(output_pipe) == 0);
    saved_stdout = dup(STDOUT_FILENO);
    assert(saved_stdout >= 0);
    assert(dup2(output_pipe[1], STDOUT_FILENO) == STDOUT_FILENO);
    close(output_pipe[1]);

    assert(renderer_render(&renderer, &state));

    assert(dup2(saved_stdout, STDOUT_FILENO) == STDOUT_FILENO);
    close(saved_stdout);
    output_length = read(output_pipe[0], output, sizeof(output) - 1);
    assert(output_length >= 0);
    output[output_length] = '\0';
    close(output_pipe[0]);

    cell = screen_buffer_get(&renderer.screen_buffer, 0, 0);
    assert(cell != NULL && cell->code_point == 'f');
    cell = screen_buffer_get(&renderer.screen_buffer, 0, 1);
    assert(cell != NULL && cell->code_point == 'g');
    cell = screen_buffer_get(&renderer.screen_buffer, 0, 2);
    assert(cell != NULL && cell->code_point == 'h');
    assert(strstr(output, "\x1b[1;2H") != NULL);

    renderer_destroy(&renderer);
    editor_state_destroy(&state);
}

/* Cursor and viewport stay in sync through long vertical and horizontal scrolls. */
static void test_cursor_and_viewport_remain_synchronized_during_scrolling(void)
{

    const size_t visible_rows = 8;
    const size_t visible_columns = 10;
    size_t row;
    size_t column;

    assert(editor_state_init(&state));

    /* Create forty logical lines, including the final empty line. */
    for (row = 0; row < 39; ++row)

        assert(document_split_line(&state.document, row, line->length));

        assert(viewport_ensure_cursor_visible(&state.viewport,

                                              state.cursor.column,
                                              visible_rows,
                                              visible_columns));
        assert(state.cursor.row - state.viewport.viewport_row < visible_rows);
    }

    for (row = state.document.line_count - 1; row > 0; --row)
    {
        assert(cursor_move_up(&state.cursor, &state.document, state.configuration.tab_width));
        assert(viewport_ensure_cursor_visible(&state.viewport,
                                              state.cursor.row,
                                              state.cursor.column,
                                              visible_rows,
                                              visible_columns));
        assert(state.cursor.row >= state.viewport.viewport_row);
        assert(state.cursor.row - state.viewport.viewport_row < visible_rows);
    }
    assert(state.cursor.row == 0);
    assert(state.viewport.viewport_row == 0);

    /* Exercise horizontal scrolling on one line longer than the viewport. */
    for (column = 0; column < 100; ++column)
    {
        assert(document_insert_character(&state.document, 0, column, 'x'));
    }
    assert(cursor_set_position(&state.cursor, &state.document, 0, 0));

    for (column = 0; column < 100; ++column)
    {
        assert(cursor_move_right(&state.cursor, &state.document));
        assert(viewport_ensure_cursor_visible(&state.viewport,
                                              state.cursor.row,
                                              state.cursor.column,
                                              visible_rows,
                                              visible_columns));
        assert(state.cursor.column >= state.viewport.viewport_column);
        assert(state.cursor.column - state.viewport.viewport_column < visible_columns);
    }

    while (state.cursor.column > 0)
    {

        assert(viewport_ensure_cursor_visible(&state.viewport,
                                              state.cursor.row,
                                              state.cursor.column,
                                              visible_rows,
                                              visible_columns));
        assert(state.cursor.column >= state.viewport.viewport_column);
        assert(state.cursor.column - state.viewport.viewport_column < visible_columns);
    }
    assert(state.viewport.viewport_column == 0);

    editor_state_destroy(&state);
}

## 1. Document invariants

Added test helpers that verify:

- The active line count does not exceed the allocated line capacity.
- The line-array pointer and capacity are consistent.
- Every active line has valid storage, with its length no greater than its capacity.
- Destroying a Document releases its storage and resets its structural fields.

New regression tests attempt invalid insertions, deletions, splits, and merges, including invalid Unicode scalar values. After each rejected operation, the tests verify that the Document remains structurally valid and retains its original contents.

## 2. Cursor invariants

The new tests verify that a Cursor remains within the Document's valid logical bounds.

They also attempt invalid Cursor position assignments, including out-of-range rows and columns. Each operation must fail without changing the Cursor's existing position, preferred logical column, or preferred visual-column state.

## 3. Viewport invariants

A new test repeatedly moves the requested Cursor position forward and backward through a larger logical area than the visible screen.

After every successful call to `viewport_ensure_cursor_visible()`, the test checks that:

- The Cursor's row is inside the visible row range.
- Its visual column is inside the visible column range.

This verifies the Viewport's behavior across repeated scrolling rather than checking only a few isolated positions.
